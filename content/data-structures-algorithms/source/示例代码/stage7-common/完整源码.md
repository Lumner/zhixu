```c
#include "tree.h"
#include <stdlib.h>
static void release(TreeNode *node)
{
    if (node != NULL) { release(node->left); release(node->right); free(node); }
}
void tree_init(BinaryTree *tree) { tree->root = NULL; tree->count = 0; }
void tree_clear(BinaryTree *tree) { release(tree->root); tree_init(tree); }
static TStatus parse(const TreeToken *tokens, size_t n, size_t *position, size_t *nodes, TreeNode **out)
{
    if (*position == n) { return T_INVALID; }
    TreeToken token = tokens[(*position)++];
    if (token.present == 0) { *out = NULL; return T_OK; }
    if (token.present != 1) { return T_INVALID; }
    if (*nodes == TREE_LIMIT) { return T_FULL; }
    TreeNode *node = malloc(sizeof *node);
    if (node == NULL) { return T_MEMORY; }
    node->value = token.value; node->left = node->right = NULL; (*nodes)++;
    TStatus status = parse(tokens, n, position, nodes, &node->left);
    if (status == T_OK) { status = parse(tokens, n, position, nodes, &node->right); }
    if (status != T_OK) { release(node); return status; }
    *out = node;
    return T_OK;
}
TStatus tree_build(BinaryTree *tree, const TreeToken *tokens, size_t count)
{
    if (tokens == NULL || count == 0) { return T_INVALID; }
    if (count > 2 * TREE_LIMIT + 1) { return T_FULL; }
    TreeNode *root = NULL; size_t position = 0, nodes = 0;
    TStatus status = parse(tokens, count, &position, &nodes, &root);
    if (status == T_OK && position != count) { status = T_INVALID; }
    if (status != T_OK) { release(root); return status; }
    tree_clear(tree); tree->root = root; tree->count = nodes;
    return T_OK;
}
static size_t height(const TreeNode *node)
{
    if (node == NULL) { return 0; }
    size_t left = height(node->left), right = height(node->right);
    return 1 + (left > right ? left : right);
}
size_t tree_height(const BinaryTree *tree) { return height(tree->root); }
```

## 38-traversal/traversal.c

完整文件：[traversal.c](../38-traversal/traversal.c)。

```c
#include "traversal.h"
static void recursive(const TreeNode *node, WalkOrder order, int *out, size_t *n)
{
    if (node == NULL) { return; }
    if (order == WALK_PRE) { out[(*n)++] = node->value; }
    recursive(node->left, order, out, n);
    if (order == WALK_IN) { out[(*n)++] = node->value; }
    recursive(node->right, order, out, n);
    if (order == WALK_POST) { out[(*n)++] = node->value; }
}
TStatus tree_walk(const BinaryTree *tree, WalkOrder order, int *out, size_t capacity, size_t *out_count)
{
    if (out_count == NULL || (tree->count != 0 && out == NULL) || order < WALK_PRE || order > WALK_POST_STACK) { return T_INVALID; }
    if (capacity < tree->count) { return T_SPACE; }
    size_t n = 0;
    if (order <= WALK_POST) { recursive(tree->root, order, out, &n); }
    else if (order == WALK_LEVEL) {
        const TreeNode *queue[TREE_LIMIT]; size_t head = 0, tail = 0;
        if (tree->root != NULL) { queue[tail++] = tree->root; }
        while (head < tail) {
            const TreeNode *node = queue[head++]; out[n++] = node->value;
            if (node->left != NULL) { queue[tail++] = node->left; }
            if (node->right != NULL) { queue[tail++] = node->right; }
        }
    }
    else if (order == WALK_PRE_STACK) {
        const TreeNode *stack[TREE_LIMIT]; size_t top = 0;
        if (tree->root != NULL) { stack[top++] = tree->root; }
        while (top != 0) {
            const TreeNode *node = stack[--top]; out[n++] = node->value;
            if (node->right != NULL) { stack[top++] = node->right; }
            if (node->left != NULL) { stack[top++] = node->left; }
        }
    }
    else if (order == WALK_IN_STACK) {
        const TreeNode *stack[TREE_LIMIT], *node = tree->root; size_t top = 0;
        while (node != NULL || top != 0) {
            while (node != NULL) { stack[top++] = node; node = node->left; }
            node = stack[--top]; out[n++] = node->value; node = node->right;
        }
    }
    else {
        typedef struct { const TreeNode *node; int expanded; } Frame;
        Frame stack[2 * TREE_LIMIT]; size_t top = 0;
        if (tree->root != NULL) { stack[top++] = (Frame){tree->root, 0}; }
        while (top != 0) {
            Frame frame = stack[--top];
            if (frame.expanded) { out[n++] = frame.node->value; }
            else {
                stack[top++] = (Frame){frame.node, 1};
                if (frame.node->right != NULL) { stack[top++] = (Frame){frame.node->right, 0}; }
                if (frame.node->left != NULL) { stack[top++] = (Frame){frame.node->left, 0}; }
            }
        }
    }
    *out_count = n;
    return T_OK;
}
```

```c
#include "bst.h"
#include <stdlib.h>
void bst_init(BST *tree) { tree->root = NULL; tree->count = 0; }
static void release(BNode *node)
{
    if (node != NULL) { release(node->left); release(node->right); free(node); }
}
void bst_clear(BST *tree) { release(tree->root); bst_init(tree); }
int bst_contains(const BST *tree, int key)
{
    const BNode *node = tree->root;
    while (node != NULL) {
        if (key == node->key) { return 1; }
        node = key < node->key ? node->left : node->right;
    }
    return 0;
}
TStatus bst_insert(BST *tree, int key)
{
    BNode *parent = NULL, *node = tree->root;
    while (node != NULL) {
        if (key == node->key) { return T_DUPLICATE; }
        parent = node; node = key < node->key ? node->left : node->right;
    }
    if (tree->count == TREE_LIMIT) { return T_FULL; }
    node = malloc(sizeof *node);
    if (node == NULL) { return T_MEMORY; }
    node->key = key; node->left = node->right = NULL;
    if (parent == NULL) { tree->root = node; }
    else if (key < parent->key) { parent->left = node; }
    else { parent->right = node; }
    tree->count++; return T_OK;
}
static BNode *erase_node(BNode *node, int key)
{
    if (key < node->key) { node->left = erase_node(node->left, key); }
    else if (key > node->key) { node->right = erase_node(node->right, key); }
    else if (node->left == NULL || node->right == NULL) {
        BNode *child = node->left != NULL ? node->left : node->right;
        free(node); return child;
    }
    else {
        BNode *successor = node->right;
        while (successor->left != NULL) { successor = successor->left; }
        node->key = successor->key;
        node->right = erase_node(node->right, successor->key);
    }
    return node;
}
TStatus bst_erase(BST *tree, int key)
{
    if (!bst_contains(tree, key)) { return T_MISSING; }
    tree->root = erase_node(tree->root, key); tree->count--; return T_OK;
}
TStatus bst_min(const BST *tree, int *out)
{
    if (out == NULL) { return T_INVALID; }
    const BNode *node = tree->root;
    if (node == NULL) { return T_EMPTY; }
    while (node->left != NULL) { node = node->left; }
    *out = node->key; return T_OK;
}
static void collect(const BNode *node, int *out, size_t *n)
{
    if (node != NULL) { collect(node->left, out, n); out[(*n)++] = node->key; collect(node->right, out, n); }
}
TStatus bst_inorder(const BST *tree, int *out, size_t capacity, size_t *out_count)
{
    if (out_count == NULL || (tree->count != 0 && out == NULL)) { return T_INVALID; }
    if (capacity < tree->count) { return T_SPACE; }
    size_t n = 0; collect(tree->root, out, &n); *out_count = n; return T_OK;
}
```

```c
#include "avl.h"
#include <stdlib.h>
void avl_init(AVL *tree) { tree->root = NULL; tree->count = 0; }
static void release(ANode *node)
{
    if (node != NULL) { release(node->left); release(node->right); free(node); }
}
void avl_clear(AVL *tree) { release(tree->root); avl_init(tree); }
static int height(const ANode *node) { return node == NULL ? 0 : node->height; }
static void refresh(ANode *node)
{
    int left = height(node->left), right = height(node->right);
    node->height = 1 + (left > right ? left : right);
}
static ANode *rotate_right(ANode *top)
{
    ANode *next = top->left; ANode *middle = next->right;
    next->right = top; top->left = middle;
    refresh(top); refresh(next); return next;
}
static ANode *rotate_left(ANode *top)
{
    ANode *next = top->right; ANode *middle = next->left;
    next->left = top; top->right = middle;
    refresh(top); refresh(next); return next;
}
int avl_contains(const AVL *tree, int key)
{
    const ANode *node = tree->root;
    while (node != NULL) {
        if (key == node->key) { return 1; }
        node = key < node->key ? node->left : node->right;
    }
    return 0;
}
static ANode *insert_node(ANode *node, int key, TStatus *status)
{
    if (node == NULL) {
        ANode *created = malloc(sizeof *created);
        if (created == NULL) { *status = T_MEMORY; return NULL; }
        *created = (ANode){key, 1, NULL, NULL}; return created;
    }
    if (key < node->key) {
        ANode *child = insert_node(node->left, key, status);
        if (*status != T_OK) { return node; }
        node->left = child;
    }
    else {
        ANode *child = insert_node(node->right, key, status);
        if (*status != T_OK) { return node; }
        node->right = child;
    }
    refresh(node);
    int balance = height(node->left) - height(node->right);
    if (balance > 1) {
        if (key > node->left->key) { node->left = rotate_left(node->left); }
        return rotate_right(node);
    }
    if (balance < -1) {
        if (key < node->right->key) { node->right = rotate_right(node->right); }
        return rotate_left(node);
    }
    return node;
}
TStatus avl_insert(AVL *tree, int key)
{
    if (avl_contains(tree, key)) { return T_DUPLICATE; }
    if (tree->count == TREE_LIMIT) { return T_FULL; }
    TStatus status = T_OK;
    tree->root = insert_node(tree->root, key, &status);
    if (status == T_OK) { tree->count++; }
    return status;
}
static void collect(const ANode *node, int *out, size_t *n)
{
    if (node != NULL) { collect(node->left, out, n); out[(*n)++] = node->key; collect(node->right, out, n); }
}
TStatus avl_inorder(const AVL *tree, int *out, size_t capacity, size_t *out_count)
{
    if (out_count == NULL || (tree->count != 0 && out == NULL)) { return T_INVALID; }
    if (capacity < tree->count) { return T_SPACE; }
    size_t n = 0; collect(tree->root, out, &n); *out_count = n; return T_OK;
}
```

```c
#include "heap.h"
static int compare(Job a, Job b)
{
    if (a.priority != b.priority) { return a.priority < b.priority ? -1 : 1; }
    if (a.id != b.id) { return a.id < b.id ? -1 : 1; }
    return 0;
}
static void swap(Job *a, Job *b) { Job saved = *a; *a = *b; *b = saved; }
static void down(MinHeap *heap, size_t root)
{
    while (root < heap->count / 2) {
        size_t child = root * 2 + 1;
        if (child + 1 < heap->count && compare(heap->data[child + 1], heap->data[child]) < 0) { child++; }
        if (compare(heap->data[root], heap->data[child]) <= 0) { break; }
        swap(&heap->data[root], &heap->data[child]); root = child;
    }
}
void heap_init(MinHeap *heap) { *heap = (MinHeap){0}; }
TStatus heap_build(MinHeap *heap, const Job *values, size_t n)
{
    if (n != 0 && values == NULL) { return T_INVALID; }
    if (n > TREE_LIMIT) { return T_FULL; }
    MinHeap next = {0}; next.count = n;
    for (size_t i = 0; i < n; i++) { next.data[i] = values[i]; }
    for (size_t i = n / 2; i > 0; i--) { down(&next, i - 1); }
    *heap = next; return T_OK;
}
TStatus heap_push(MinHeap *heap, Job job)
{
    if (heap->count == TREE_LIMIT) { return T_FULL; }
    size_t child = heap->count++; heap->data[child] = job;
    while (child > 0) {
        size_t parent = (child - 1) / 2;
        if (compare(heap->data[parent], heap->data[child]) <= 0) { break; }
        swap(&heap->data[parent], &heap->data[child]); child = parent;
    }
    return T_OK;
}
TStatus heap_peek(const MinHeap *heap, Job *out)
{
    if (out == NULL) { return T_INVALID; }
    if (heap->count == 0) { return T_EMPTY; }
    *out = heap->data[0]; return T_OK;
}
TStatus heap_pop(MinHeap *heap, Job *out)
{
    if (out == NULL) { return T_INVALID; }
    if (heap->count == 0) { return T_EMPTY; }
    Job saved = heap->data[0]; heap->count--;
    if (heap->count != 0) { heap->data[0] = heap->data[heap->count]; down(heap, 0); }
    *out = saved; return T_OK;
}
```

```c
#include "heapsort.h"
#include "sort_internal.h"
static void down_max(SItem *a, size_t root, size_t n, SortStats *stats)
{
    while (root < n / 2) {
        size_t child = root * 2 + 1;
        if (child + 1 < n && sort_compare(a[child + 1].key, a[child].key, stats) > 0) { child++; }
        if (sort_compare(a[root].key, a[child].key, stats) >= 0) { break; }
        sort_swap(a, root, child, stats); root = child;
    }
}
SortStatus heap_sort(SItem *a, size_t n, SortStats *out)
{
    SortStatus status = sort_validate(a, n, out);
    if (status != S_OK) { return status; }
    SortStats stats = {0, 0};
    for (size_t i = n / 2; i > 0; i--) { down_max(a, i - 1, n, &stats); }
    for (size_t end = n; end > 1; end--) {
        sort_swap(a, 0, end - 1, &stats); down_max(a, 0, end - 1, &stats);
    }
    *out = stats; return S_OK;
}
```

```c
#include "huffman.h"
#include <string.h>
void huff_init(Huffman *tree) { *tree = (Huffman){0}; tree->root = -1; }
static int minimum(const Huffman *tree, const int *active)
{
    int best = -1;
    for (size_t i = 0; i < tree->count; i++) {
        if (active[i] && (best < 0 || tree->nodes[i].weight < tree->nodes[best].weight)) { best = (int)i; }
    }
    return best;
}
TStatus huff_build(Huffman *tree, const HuffSymbol *symbols, size_t n)
{
    if (n != 0 && symbols == NULL) { return T_INVALID; }
    if (n > HUFF_SYMBOLS) { return T_FULL; }
    for (size_t i = 0; i < n; i++) {
        if (symbols[i].weight == 0 || symbols[i].weight > 1000) { return T_INVALID; }
        for (size_t j = 0; j < i; j++) { if (symbols[j].symbol == symbols[i].symbol) { return T_DUPLICATE; } }
    }
    Huffman next; huff_init(&next); int active[HUFF_NODES] = {0};
    next.symbols = next.count = n;
    for (size_t i = 0; i < n; i++) { next.nodes[i] = (HuffNode){symbols[i].weight,-1,-1,-1,symbols[i].symbol}; active[i] = 1; }
    for (size_t remaining = n; remaining > 1; remaining--) {
        int left = minimum(&next, active); active[left] = 0;
        int right = minimum(&next, active); active[right] = 0;
        int parent = (int)next.count++;
        next.nodes[parent] = (HuffNode){next.nodes[left].weight + next.nodes[right].weight,left,right,-1,0};
        next.nodes[left].parent = next.nodes[right].parent = parent; active[parent] = 1;
    }
    next.root = n == 0 ? -1 : (int)next.count - 1;
    *tree = next; return T_OK;
}
static size_t code_at(const Huffman *tree, int leaf, char *code)
{
    if (tree->symbols == 1) { code[0] = '0'; code[1] = '\0'; return 1; }
    char reverse[HUFF_SYMBOLS]; size_t n = 0;
    while (tree->nodes[leaf].parent >= 0) {
        int parent = tree->nodes[leaf].parent;
        reverse[n++] = tree->nodes[parent].left == leaf ? '0' : '1'; leaf = parent;
    }
    for (size_t i = 0; i < n; i++) { code[i] = reverse[n - 1 - i]; }
    code[n] = '\0'; return n;
}
TStatus huff_code(const Huffman *tree, unsigned char symbol, char *out, size_t capacity)
{
    if (out == NULL) { return T_INVALID; }
    for (size_t i = 0; i < tree->symbols; i++) {
        if (tree->nodes[i].symbol == symbol) {
            char code[HUFF_SYMBOLS + 1]; size_t n = code_at(tree, (int)i, code);
            if (capacity <= n) { return T_SPACE; }
            memcpy(out, code, n + 1); return T_OK;
        }
    }
    return T_MISSING;
}
TStatus huff_encode(const Huffman *tree, const unsigned char *text, size_t n, char *out, size_t capacity, size_t *bits)
{
    if (out == NULL || bits == NULL || (n != 0 && text == NULL)) { return T_INVALID; }
    if (n > HUFF_TEXT_MAX) { return T_FULL; }
    char candidate[HUFF_BITS_MAX + 1]; size_t used = 0;
    for (size_t i = 0; i < n; i++) {
        char code[HUFF_SYMBOLS + 1];
        if (huff_code(tree, text[i], code, sizeof code) != T_OK) { return T_INVALID; }
        for (size_t j = 0; code[j] != '\0'; j++) { candidate[used++] = code[j]; }
    }
    if (capacity <= used) { return T_SPACE; }
    candidate[used] = '\0'; memcpy(out, candidate, used + 1); *bits = used; return T_OK;
}
TStatus huff_decode(const Huffman *tree, const char *bits, size_t n, unsigned char *out, size_t capacity, size_t *bytes)
{
    if (bytes == NULL || (n != 0 && bits == NULL) || (capacity != 0 && out == NULL)) { return T_INVALID; }
    if (n > HUFF_BITS_MAX) { return T_FULL; }
    unsigned char candidate[HUFF_TEXT_MAX]; size_t used = 0;
    if (tree->root < 0) { if (n != 0) { return T_INVALID; } }
    else {
        int cursor = tree->root;
        for (size_t i = 0; i < n; i++) {
            if (bits[i] != '0' && bits[i] != '1') { return T_INVALID; }
            if (tree->symbols == 1) { if (bits[i] != '0') { return T_INVALID; } }
            else { cursor = bits[i] == '0' ? tree->nodes[cursor].left : tree->nodes[cursor].right; }
            if (tree->nodes[cursor].left < 0) {
                if (used == HUFF_TEXT_MAX) { return T_FULL; }
                candidate[used++] = tree->nodes[cursor].symbol; cursor = tree->root;
            }
        }
        if (cursor != tree->root) { return T_INVALID; }
    }
    if (capacity < used) { return T_SPACE; }
    if (used != 0) { memcpy(out, candidate, used); }
    *bytes = used; return T_OK;
}
unsigned long huff_cost(const Huffman *tree)
{
    unsigned long total = 0;
    for (size_t i = 0; i < tree->symbols; i++) {
        char code[HUFF_SYMBOLS + 1]; size_t n = code_at(tree, (int)i, code);
        total += (unsigned long)tree->nodes[i].weight * n;
    }
    return total;
}
```
