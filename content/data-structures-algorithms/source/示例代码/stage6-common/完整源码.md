```c
#include "sort_internal.h"
SortStatus sort_validate(const SItem *a, size_t n, const SortStats *out)
{
    if (out == NULL || (n != 0 && a == NULL)) { return S_INVALID; }
    return n > SORT_MAX ? S_RANGE : S_OK;
}
SortStatus sort_domain(const SItem *a, size_t n)
{
    for (size_t i = 0; i < n; i++) {
        if (a[i].key < SORT_LOW || a[i].key > SORT_HIGH) { return S_RANGE; }
    }
    return S_OK;
}
int sort_compare(int left, int right, SortStats *stats)
{
    stats->comparisons++;
    if (left < right) { return -1; }
    if (left > right) { return 1; }
    return 0;
}
void sort_write(SItem *a, size_t i, SItem value, SortStats *stats)
{
    a[i] = value;
    stats->writes++;
}
void sort_swap(SItem *a, size_t i, size_t j, SortStats *stats)
{
    if (i != j) {
        SItem saved = a[i];
        sort_write(a, i, a[j], stats);
        sort_write(a, j, saved, stats);
    }
}
```

## 31-basic-sort/basic.c

完整文件：[basic.c](../31-basic-sort/basic.c)。

```c
#include "sort_internal.h"
SortStatus selection_sort(SItem *a, size_t n, SortStats *out)
{
    SortStatus status = sort_validate(a, n, out);
    if (status != S_OK) { return status; }
    SortStats stats = {0, 0};
    for (size_t i = 0; i + 1 < n; i++) {
        size_t best = i;
        for (size_t j = i + 1; j < n; j++) {
            if (sort_compare(a[j].key, a[best].key, &stats) < 0) { best = j; }
        }
        sort_swap(a, i, best, &stats);
    }
    *out = stats;
    return S_OK;
}
SortStatus bubble_sort(SItem *a, size_t n, SortStats *out)
{
    SortStatus status = sort_validate(a, n, out);
    if (status != S_OK) { return status; }
    SortStats stats = {0, 0};
    for (size_t end = n; end > 1; end--) {
        int changed = 0;
        for (size_t j = 1; j < end; j++) {
            if (sort_compare(a[j - 1].key, a[j].key, &stats) > 0) {
                sort_swap(a, j - 1, j, &stats);
                changed = 1;
            }
        }
        if (!changed) { break; }
    }
    *out = stats;
    return S_OK;
}
```

## 32-insertion-shell/insertion_shell.c

完整文件：[insertion_shell.c](../32-insertion-shell/insertion_shell.c)。

```c
#include "sort_internal.h"
SortStatus insertion_sort(SItem *a, size_t n, SortStats *out)
{
    SortStatus status = sort_validate(a, n, out);
    if (status != S_OK) { return status; }
    SortStats stats = {0, 0};
    for (size_t i = 1; i < n; i++) {
        SItem saved = a[i];
        size_t j = i;
        while (j > 0 && sort_compare(a[j - 1].key, saved.key, &stats) > 0) {
            sort_write(a, j, a[j - 1], &stats);
            j--;
        }
        sort_write(a, j, saved, &stats);
    }
    *out = stats;
    return S_OK;
}
SortStatus shell_sort(SItem *a, size_t n, SortStats *out)
{
    SortStatus status = sort_validate(a, n, out);
    if (status != S_OK) { return status; }
    SortStats stats = {0, 0};
    for (size_t gap = n / 2; gap > 0; gap /= 2) {
        for (size_t i = gap; i < n; i++) {
            SItem saved = a[i];
            size_t j = i;
            while (j >= gap && sort_compare(a[j - gap].key, saved.key, &stats) > 0) {
                sort_write(a, j, a[j - gap], &stats);
                j -= gap;
            }
            sort_write(a, j, saved, &stats);
        }
    }
    *out = stats;
    return S_OK;
}
```

## 33-merge/merge.c

完整文件：[merge.c](../33-merge/merge.c)。

```c
#include "sort_internal.h"
#include <stdlib.h>
static void merge_range(SItem *a, SItem *work, size_t lo, size_t hi, SortStats *stats)
{
    if (hi - lo < 2) { return; }
    size_t mid = lo + (hi - lo) / 2;
    merge_range(a, work, lo, mid, stats);
    merge_range(a, work, mid, hi, stats);
    size_t left = lo, right = mid, dest = lo;
    while (left < mid && right < hi) {
        if (sort_compare(a[left].key, a[right].key, stats) <= 0) {
            sort_write(work, dest++, a[left++], stats);
        }
        else { sort_write(work, dest++, a[right++], stats); }
    }
    while (left < mid) { sort_write(work, dest++, a[left++], stats); }
    while (right < hi) { sort_write(work, dest++, a[right++], stats); }
    for (size_t i = lo; i < hi; i++) { sort_write(a, i, work[i], stats); }
}
SortStatus merge_sort(SItem *a, size_t n, SortStats *out)
{
    SortStatus status = sort_validate(a, n, out);
    if (status != S_OK) { return status; }
    SortStats stats = {0, 0};
    if (n > 1) {
        SItem *work = malloc(n * sizeof *work);
        if (work == NULL) { return S_MEMORY; }
        merge_range(a, work, 0, n, &stats);
        free(work);
    }
    *out = stats;
    return S_OK;
}
```

## 34-quick/quick.c

完整文件：[quick.c](../34-quick/quick.c)。

```c
#include "sort_internal.h"
static void quick_range(SItem *a, size_t lo, size_t hi, SortStats *stats)
{
    while (hi - lo > 1) {
        int pivot = a[lo + (hi - lo) / 2].key;
        size_t lt = lo, i = lo, gt = hi;
        while (i < gt) {
            int order = sort_compare(a[i].key, pivot, stats);
            if (order < 0) { sort_swap(a, lt, i, stats); lt++; i++; }
            else if (order > 0) { gt--; sort_swap(a, i, gt, stats); }
            else { i++; }
        }
        if (lt - lo < hi - gt) {
            quick_range(a, lo, lt, stats);
            lo = gt;
        }
        else {
            quick_range(a, gt, hi, stats);
            hi = lt;
        }
    }
}
SortStatus quick_sort(SItem *a, size_t n, SortStats *out)
{
    SortStatus status = sort_validate(a, n, out);
    if (status != S_OK) { return status; }
    SortStats stats = {0, 0};
    quick_range(a, 0, n, &stats);
    *out = stats;
    return S_OK;
}
```

## 35-distribution/distribution.c

完整文件：[distribution.c](../35-distribution/distribution.c)。

```c
#include "sort_internal.h"
#include <stdlib.h>
#define KEY_RANGE (SORT_HIGH - SORT_LOW + 1)
#define BUCKET_COUNT 10
#define BUCKET_WIDTH 200
static void positions(const size_t *counts, size_t *next, size_t k)
{
    size_t total = 0;
    for (size_t i = 0; i < k; i++) { next[i] = total; total += counts[i]; }
}
static SortStatus prepare(const SItem *a, size_t n, const SortStats *out)
{
    SortStatus status = sort_validate(a, n, out);
    return status == S_OK ? sort_domain(a, n) : status;
}
SortStatus counting_sort(SItem *a, size_t n, SortStats *out)
{
    SortStatus status = prepare(a, n, out);
    if (status != S_OK) { return status; }
    SortStats stats = {0, 0};
    if (n > 1) {
        SItem *work = malloc(n * sizeof *work);
        if (work == NULL) { return S_MEMORY; }
        size_t counts[KEY_RANGE] = {0}, next[KEY_RANGE];
        for (size_t i = 0; i < n; i++) { counts[a[i].key - SORT_LOW]++; }
        positions(counts, next, KEY_RANGE);
        for (size_t i = 0; i < n; i++) {
            size_t key = (size_t)(a[i].key - SORT_LOW);
            sort_write(work, next[key]++, a[i], &stats);
        }
        for (size_t i = 0; i < n; i++) { sort_write(a, i, work[i], &stats); }
        free(work);
    }
    *out = stats;
    return S_OK;
}
SortStatus bucket_sort(SItem *a, size_t n, SortStats *out)
{
    SortStatus status = prepare(a, n, out);
    if (status != S_OK) { return status; }
    SortStats stats = {0, 0};
    if (n > 1) {
        SItem *work = malloc(n * sizeof *work);
        if (work == NULL) { return S_MEMORY; }
        size_t counts[BUCKET_COUNT] = {0}, starts[BUCKET_COUNT], next[BUCKET_COUNT];
        for (size_t i = 0; i < n; i++) { counts[(a[i].key - SORT_LOW) / BUCKET_WIDTH]++; }
        positions(counts, starts, BUCKET_COUNT);
        for (size_t b = 0; b < BUCKET_COUNT; b++) { next[b] = starts[b]; }
        for (size_t i = 0; i < n; i++) {
            size_t b = (size_t)((a[i].key - SORT_LOW) / BUCKET_WIDTH);
            sort_write(work, next[b]++, a[i], &stats);
        }
        for (size_t b = 0; b < BUCKET_COUNT; b++) {
            size_t lo = starts[b], hi = lo + counts[b];
            for (size_t i = lo + 1; i < hi; i++) {
                SItem saved = work[i]; size_t j = i;
                while (j > lo && sort_compare(work[j - 1].key, saved.key, &stats) > 0) {
                    sort_write(work, j, work[j - 1], &stats); j--;
                }
                sort_write(work, j, saved, &stats);
            }
        }
        for (size_t i = 0; i < n; i++) { sort_write(a, i, work[i], &stats); }
        free(work);
    }
    *out = stats;
    return S_OK;
}
SortStatus radix_sort(SItem *a, size_t n, SortStats *out)
{
    SortStatus status = prepare(a, n, out);
    if (status != S_OK) { return status; }
    SortStats stats = {0, 0};
    if (n > 1) {
        SItem *work = malloc(n * sizeof *work);
        if (work == NULL) { return S_MEMORY; }
        for (int factor = 1; factor <= 1000; factor *= 10) {
            size_t counts[10] = {0}, next[10];
            for (size_t i = 0; i < n; i++) { counts[((a[i].key - SORT_LOW) / factor) % 10]++; }
            positions(counts, next, 10);
            for (size_t i = 0; i < n; i++) {
                size_t digit = (size_t)(((a[i].key - SORT_LOW) / factor) % 10);
                sort_write(work, next[digit]++, a[i], &stats);
            }
            for (size_t i = 0; i < n; i++) { sort_write(a, i, work[i], &stats); }
        }
        free(work);
    }
    *out = stats;
    return S_OK;
}
```

```c
#include "experiment.h"
#include <string.h>
const char *algorithm_name(size_t algorithm)
{
    static const char *names[] = {"selection", "bubble", "insertion", "shell", "merge", "quick", "counting", "bucket", "radix"};
    return algorithm < ALGORITHM_COUNT ? names[algorithm] : "invalid";
}
const char *dataset_name(size_t dataset)
{
    static const char *names[] = {"sorted", "reverse", "equal", "nearly", "mixed"};
    return dataset < DATASET_COUNT ? names[dataset] : "invalid";
}
int algorithm_is_stable(size_t algorithm)
{
    return algorithm == 1 || algorithm == 2 || algorithm == 4 || algorithm >= 6;
}
SortStatus experiment_run(size_t algorithm, SItem *a, size_t n, SortStats *out)
{
    switch (algorithm) {
    case 0: return selection_sort(a, n, out);
    case 1: return bubble_sort(a, n, out);
    case 2: return insertion_sort(a, n, out);
    case 3: return shell_sort(a, n, out);
    case 4: return merge_sort(a, n, out);
    case 5: return quick_sort(a, n, out);
    case 6: return counting_sort(a, n, out);
    case 7: return bucket_sort(a, n, out);
    case 8: return radix_sort(a, n, out);
    default: return S_INVALID;
    }
}
void experiment_data(size_t dataset, SItem *a, size_t n)
{
    for (size_t i = 0; i < n; i++) {
        int key = (int)i;
        if (dataset == 1) { key = (int)(n - 1 - i); }
        else if (dataset == 2) { key = 5; }
        else if (dataset == 4) { key = (int)((i * 17) % 23) - 11; }
        a[i].key = key; a[i].tag = (int)i;
    }
    if (dataset == 3 && n > 1) {
        SItem saved = a[0]; a[0] = a[1]; a[1] = saved;
    }
}
int experiment_check(const SItem *before, const SItem *after, size_t n, int require_stable)
{
    if (n > SORT_MAX) { return 0; }
    int seen[SORT_MAX] = {0};
    size_t previous_position = 0;
    for (size_t i = 0; i < n; i++) {
        if (i > 0 && after[i - 1].key > after[i].key) { return 0; }
        size_t position = 0;
        while (position < n && (before[position].key != after[i].key || before[position].tag != after[i].tag || seen[position])) { position++; }
        if (position == n) { return 0; }
        seen[position] = 1;
        if (require_stable && i > 0 && after[i - 1].key == after[i].key && previous_position > position) { return 0; }
        previous_position = position;
    }
    return 1;
}
```

## stage6-lab/main.c

完整文件：[main.c](../stage6-lab/main.c)。

```c
#include "experiment.h"
#include <stdio.h>
#include <string.h>
int main(int argc, char **argv)
{
    if (argc != 3) { fputs("usage: sort-lab algorithm|all dataset\n", stderr); return 1; }
    size_t first = 0, last = ALGORITHM_COUNT, dataset = 0;
    if (strcmp(argv[1], "all") != 0) {
        while (first < ALGORITHM_COUNT && strcmp(argv[1], algorithm_name(first)) != 0) { first++; }
        if (first == ALGORITHM_COUNT) { fputs("unknown algorithm\n", stderr); return 1; }
        last = first + 1;
    }
    while (dataset < DATASET_COUNT && strcmp(argv[2], dataset_name(dataset)) != 0) { dataset++; }
    if (dataset == DATASET_COUNT) { fputs("unknown dataset\n", stderr); return 1; }
    SItem original[32], a[32];
    experiment_data(dataset, original, 32);
    for (size_t k = first; k < last; k++) {
        memcpy(a, original, sizeof a); SortStats stats;
        if (experiment_run(k, a, 32, &stats) != S_OK || !experiment_check(original, a, 32, algorithm_is_stable(k))) {
            fputs("experiment failed\n", stderr); return 1;
        }
        printf("%s %s n=32 comparisons=%zu writes=%zu verified=1\n", algorithm_name(k), dataset_name(dataset), stats.comparisons, stats.writes);
    }
    return 0;
}
```

```c
{
    if (catalog->count > RECORD_LIMIT) { return CAT_LIMIT; }
    Catalog serial;
    catalog_init(&serial);
    CatStatus status = CAT_OK;
    for (size_t i = 0; i < catalog->count && status == CAT_OK; i++) {
        Record row = {0};
        row.id = catalog->records[i].id;
        row.minutes = catalog->records[i].minutes;
        strcpy(row.title, catalog->records[i].title);
        status = catalog_add(&serial, &row);
    }
    if (status == CAT_OK) { status = catalog_save(out, &serial); }
    catalog_free(&serial);
    return status;
}
```

```c
#include "io.h"
#include "view.h"
#include "matching.h"
#include <string.h>
static void print_record(const IRecord *record)
{
    printf("id=%d minutes=%d title=%s\n", record->id, record->minutes, record->title);
}
int main(int argc, char **argv)
{
    if (argc < 3) { fputs("usage: catalog-v3 view|save|find|title|remove input [arguments]\n", stderr); return 1; }
    const char *mode = argv[1];
    int view = strcmp(mode, "view") == 0, save = strcmp(mode, "save") == 0;
    int find = strcmp(mode, "find") == 0, title = strcmp(mode, "title") == 0, remove = strcmp(mode, "remove") == 0;
    if (!((view && argc == 3) || (save && (argc == 3 || argc == 5)) || ((find || title || remove) && argc == 4))) {
        fputs("usage: catalog-v3 view|save|find|title|remove input [arguments]\n", stderr); return 1;
    }
    int id = 0, minutes = 0;
    if (((find || remove || (save && argc == 5)) && !parse_number(argv[3], 9999, &id)) ||
        (save && argc == 5 && !parse_number(argv[4], 1440, &minutes)) ||
        (title && strlen(argv[3]) > MATCH_PATTERN_MAX)) {
        fputs("v3: invalid arguments\n", stderr); return 1;
    }
    FILE *in = fopen(argv[2], "rb");
    if (in == NULL) { fputs("v3: open failed\n", stderr); return 1; }
    IndexedCatalog catalog;
    indexed_init(&catalog);
    CatStatus status = v3_load(in, &catalog);
    if (fclose(in) != 0 && status == CAT_OK) { status = CAT_IO; }
    if (status == CAT_OK && save && argc == 5) { status = v3_hash_status(indexed_set_minutes(&catalog, id, minutes)); }
    if (status == CAT_OK && remove) { status = v3_hash_status(indexed_remove(&catalog, id)); }
    if (status == CAT_OK) {
        if (save || remove) { status = v3_save(stdout, &catalog); }
        else if (view) {
            IRecord sorted[H_MAX_ITEMS]; SortStats stats;
            SortStatus sorted_status = catalog_view(&catalog, sorted, H_MAX_ITEMS, &stats);
            if (sorted_status != S_OK) { status = sorted_status == S_MEMORY ? CAT_MEMORY : CAT_INVALID; }
            else { for (size_t i = 0; i < catalog.count; i++) { print_record(&sorted[i]); } }
        }
        else if (find) {
            IRecord found;
            status = v3_hash_status(indexed_find(&catalog, id, &found));
            if (status == CAT_OK) { print_record(&found); }
        }
        else {
            for (size_t i = 0; i < catalog.count; i++) {
                Match match;
                (void)kmp_match(catalog.records[i].title, strlen(catalog.records[i].title), argv[3], strlen(argv[3]), &match);
                if (match.found) { print_record(&catalog.records[i]); }
            }
        }
    }
    if (status == CAT_OK && (ferror(stdout) || fflush(stdout) != 0)) { status = CAT_IO; }
    indexed_clear(&catalog);
    if (status != CAT_OK) { fprintf(stderr, "v3: %s\n", catalog_status_name(status)); return 1; }
    return 0;
}
```

```c
#include "view.h"
SortStatus catalog_view(const IndexedCatalog *catalog, IRecord *out, size_t capacity, SortStats *stats)
{
    if (stats == NULL || (catalog->count != 0 && out == NULL)) { return S_INVALID; }
    if (capacity < catalog->count) { return S_RANGE; }
    SItem order[H_MAX_ITEMS];
    for (size_t i = 0; i < catalog->count; i++) {
        order[i].key = catalog->records[i].minutes;
        order[i].tag = (int)i;
    }
    SortStats measured;
    SortStatus status = merge_sort(order, catalog->count, &measured);
    if (status != S_OK) { return status; }
    for (size_t i = 0; i < catalog->count; i++) { out[i] = catalog->records[order[i].tag]; }
    *stats = measured;
    return S_OK;
}
```
