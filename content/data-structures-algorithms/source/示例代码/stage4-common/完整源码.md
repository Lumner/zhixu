```c
        return DS_RANGE;
    }

    return DS_OK;

DsStatus seq_insert(Sequence *list, size_t position, int value)
{
    if (position > list->count) {
        return DS_RANGE;
    }
    if (list->count == SEQ_CAPACITY) {
        return DS_FULL;
    }
    for (size_t i = list->count; i > position; i--) {
        list->data[i] = list->data[i - 1];
    }
    list->data[position] = value;
    list->count++;
    return DS_OK;
}

DsStatus seq_erase(Sequence *list, size_t position, int *out)
{
    if (out == NULL) {
        return DS_INVALID;
    }
    if (position >= list->count) {
        return DS_RANGE;
    }
    int saved = list->data[position];
    for (size_t i = position; i + 1 < list->count; i++) {
        list->data[i] = list->data[i + 1];
    }
    list->count--;
    *out = saved;
    return DS_OK;
}
```

```c
        printf("exact count=%zu capacity=%zu copy_model=%zu\n", exact.count, exact.capacity, exact_copies);

    vec_clear(&growing);
    vec_clear(&exact);
```

```c
#include "vector.h"
#include <stdint.h>
#include <stdlib.h>

void vec_init(Vector *list)
{
    list->data = NULL;
    list->count = list->capacity = 0;
}

void vec_clear(Vector *list) {
    free(list->data);
    vec_init(list);
}

DsStatus vec_reserve(Vector *list, size_t capacity)
{
    if (capacity > DS_MAX_ITEMS) {
        return DS_FULL;
    }
    if (capacity <= list->capacity) {
        return DS_OK;
    }
    if (capacity > SIZE_MAX / sizeof *list->data) {
        return DS_MEMORY;
    }
    int *grown = realloc(list->data, capacity * sizeof *list->data);
    if (grown == NULL) {
        return DS_MEMORY;
    }
    list->data = grown;
    list->capacity = capacity;
    return DS_OK;
}

DsStatus vec_get(const Vector *list, size_t position, int *out)
{
    if (out == NULL) {
        return DS_INVALID;
    }
    if (position >= list->count) {
        return DS_RANGE;
    }
    *out = list->data[position];
    return DS_OK;
}

DsStatus vec_set(Vector *list, size_t position, int value)
{
    if (position >= list->count) {
        return DS_RANGE;
    }
    list->data[position] = value;
    return DS_OK;
}

DsStatus vec_insert(Vector *list, size_t position, int value)
{
    if (position > list->count) {
        return DS_RANGE;
    }
    if (list->count == DS_MAX_ITEMS) {
        return DS_FULL;
    }
    if (list->count == list->capacity) {
        size_t next = list->capacity == 0 ? 4 : list->capacity * 2;
        if (next > DS_MAX_ITEMS) {
            next = DS_MAX_ITEMS;
        }
        DsStatus status = vec_reserve(list, next);
        if (status != DS_OK) {
            return status;
        }
    }
    for (size_t i = list->count; i > position; i--) {
        list->data[i] = list->data[i - 1];
    }
    list->data[position] = value;
    list->count++;
    return DS_OK;
}

DsStatus vec_erase(Vector *list, size_t position, int *out)
{
    if (out == NULL) {
        return DS_INVALID;
    }
    if (position >= list->count) {
        return DS_RANGE;
    }
    int saved = list->data[position];
    for (size_t i = position; i + 1 < list->count; i++) {
        list->data[i] = list->data[i + 1];
    }
    list->count--;
    *out = saved;
    return DS_OK;
}
```

## 23-singly/singly.c

完整文件：[singly.c](../23-singly/singly.c)。

```c
#include "singly.h"
#include <stdlib.h>

void slist_init(SList *list) {
    list->head.next = NULL;
    list->count = 0;
}

void slist_clear(SList *list)
{
    SNode *node = list->head.next;
    while (node != NULL) {
        SNode *next = node->next;
        free(node);
        node = next;
    }
    slist_init(list);
}

static SNode *before(SList *list, size_t position)
{
    SNode *previous = &list->head;
    for (size_t i = 0; i < position; i++) {
        previous = previous->next;
    }
    return previous;
}

DsStatus slist_get(const SList *list, size_t position, int *out)
{
    if (out == NULL) {
        return DS_INVALID;
    }
    if (position >= list->count) {
        return DS_RANGE;
    }
    const SNode *node = list->head.next;
    for (size_t i = 0; i < position; i++) {
        node = node->next;
    }
    *out = node->value;
    return DS_OK;
}

DsStatus slist_set(SList *list, size_t position, int value)
{
    if (position >= list->count) {
        return DS_RANGE;
    }
    before(list, position)->next->value = value;
    return DS_OK;
}

DsStatus slist_insert(SList *list, size_t position, int value)
{
    if (position > list->count) {
        return DS_RANGE;
    }
    if (list->count == DS_MAX_ITEMS) {
        return DS_FULL;
    }
    SNode *previous = before(list, position);
    SNode *node = malloc(sizeof *node);
    if (node == NULL) {
        return DS_MEMORY;
    }
    node->value = value;
    node->next = previous->next;
    previous->next = node;
    list->count++;
    return DS_OK;
}

DsStatus slist_erase(SList *list, size_t position, int *out)
{
    if (out == NULL) {
        return DS_INVALID;
    }
    if (position >= list->count) {
        return DS_RANGE;
    }
    SNode *previous = before(list, position);
    SNode *node = previous->next;
    int saved = node->value;
    previous->next = node->next;
    free(node);
    list->count--;
    *out = saved;
    return DS_OK;
}
```

## 24-doubly/doubly.c

完整文件：[doubly.c](../24-doubly/doubly.c)。

```c
#include "doubly.h"
#include <stdlib.h>

void dlist_init(DList *list)
{
    list->root.next = list->root.previous = &list->root;
    list->count = 0;
}

void dlist_clear(DList *list)
{
    DNode *node = list->root.next;
    while (node != &list->root) {
        DNode *next = node->next;
        free(node);
        node = next;
    }
    dlist_init(list);
}

static DNode *node_at(const DList *list, size_t position)
{
    DNode *node;
    if (position < list->count / 2) {
        node = list->root.next;
        for (size_t i = 0; i < position; i++) {
            node = node->next;
        }
    } else {
        node = list->root.previous;
        for (size_t i = list->count - 1; i > position; i--) {
            node = node->previous;
        }
    }
    return node;
}

DsStatus dlist_get(const DList *list, size_t position, int *out)
{
    if (out == NULL) {
        return DS_INVALID;
    }
    if (position >= list->count) {
        return DS_RANGE;
    }
    *out = node_at(list, position)->value;
    return DS_OK;
}

DsStatus dlist_set(DList *list, size_t position, int value)
{
    if (position >= list->count) {
        return DS_RANGE;
    }
    node_at(list, position)->value = value;
    return DS_OK;
}

DsStatus dlist_insert(DList *list, size_t position, int value)
{
    if (position > list->count) {
        return DS_RANGE;
    }
    if (list->count == DS_MAX_ITEMS) {
        return DS_FULL;
    }
    DNode *next = position == list->count ? &list->root : node_at(list, position);
    DNode *node = malloc(sizeof *node);
    if (node == NULL) {
        return DS_MEMORY;
    }
    node->value = value;
    node->previous = next->previous;
    node->next = next;
    next->previous->next = node;
    next->previous = node;
    list->count++;
    return DS_OK;
}

DsStatus dlist_erase(DList *list, size_t position, int *out)
{
    if (out == NULL) {
        return DS_INVALID;
    }
    if (position >= list->count) {
        return DS_RANGE;
    }
    DNode *node = node_at(list, position);
    int saved = node->value;
    node->previous->next = node->next;
    node->next->previous = node->previous;
    free(node);
    list->count--;
    *out = saved;
    return DS_OK;
}
```

完整文件：[brackets.c](../25-stack/brackets.c)。

```c
#include "brackets.h"
#include "stack.h"
#include <string.h>

BracketStatus brackets_check(const char *text, size_t *error_position)
{
    if (text == NULL || error_position == NULL) {
        return B_INVALID;
    }
    size_t length = strlen(text);
    if (length > 256) {
        *error_position = 256;
        return B_LENGTH;
    }
    ArrayStack pending;
    stack_init(&pending);
    for (size_t i = 0; i < length; i++) {
        char ch = text[i];
        if (ch == '(' || ch == '[' || ch == '{') {
            if (stack_push(&pending, (int)i) != DS_OK) {
                *error_position = i;
                return B_DEPTH;
            }
        } else if (ch == ')' || ch == ']' || ch == '}') {
            int opening = 0;
            if (stack_peek(&pending, &opening) != DS_OK ||
                !((text[opening] == '(' && ch == ')') ||
                  (text[opening] == '[' && ch == ']') ||
                  (text[opening] == '{' && ch == '}'))) {
                *error_position = i;
                return B_MISMATCH;
            }
            if (stack_pop(&pending, &opening) != DS_OK) {
                return B_INVALID;
            }
        }
    }
    if (pending.count > 0) {
        int earliest = 0;
        while (stack_pop(&pending, &earliest) == DS_OK) {
        }
        *error_position = (size_t)earliest;
        return B_UNCLOSED;
    }
    return B_OK;
}

const char *brackets_status_name(BracketStatus status)
{
    switch (status) {
    case B_OK: return "ok";
    case B_MISMATCH: return "mismatch";
    case B_UNCLOSED: return "unclosed";
    case B_DEPTH: return "depth";
    case B_LENGTH: return "length";
    default: return "invalid";
    }
}
```

## 25-stack/stack.c

完整文件：[stack.c](../25-stack/stack.c)。

```c
#include "stack.h"
#include <stdlib.h>

void stack_init(ArrayStack *stack) {
    stack->count = 0;
}
DsStatus stack_push(ArrayStack *stack, int value)
{
    if (stack->count == STACK_CAPACITY) {
        return DS_FULL;
    }
    stack->data[stack->count] = value;
    stack->count++;
    return DS_OK;
}
DsStatus stack_pop(ArrayStack *stack, int *out)
{
    if (out == NULL) {
        return DS_INVALID;
    }
    if (stack->count == 0) {
        return DS_EMPTY;
    }
    *out = stack->data[stack->count - 1];
    stack->count--;
    return DS_OK;
}
DsStatus stack_peek(const ArrayStack *stack, int *out)
{
    if (out == NULL) {
        return DS_INVALID;
    }
    if (stack->count == 0) {
        return DS_EMPTY;
    }
    *out = stack->data[stack->count - 1];
    return DS_OK;
}
void lstack_init(LinkedStack *stack) {
    stack->top = NULL;
    stack->count = 0;
}
void lstack_clear(LinkedStack *stack)
{
    StackNode *node = stack->top;
    while (node != NULL) {
        StackNode *next = node->next;
        free(node);
        node = next;
    }
    lstack_init(stack);
}
DsStatus lstack_push(LinkedStack *stack, int value)
{
    if (stack->count == DS_MAX_ITEMS) {
        return DS_FULL;
    }
    StackNode *node = malloc(sizeof *node);
    if (node == NULL) {
        return DS_MEMORY;
    }
    node->value = value;
    node->next = stack->top;
    stack->top = node;
    stack->count++;
    return DS_OK;
}
DsStatus lstack_pop(LinkedStack *stack, int *out)
{
    if (out == NULL) {
        return DS_INVALID;
    }
    if (stack->count == 0) {
        return DS_EMPTY;
    }
    StackNode *node = stack->top;
    int saved = node->value;
    stack->top = node->next;
    free(node);
    stack->count--;
    *out = saved;
    return DS_OK;
}
DsStatus lstack_peek(const LinkedStack *stack, int *out)
{
    if (out == NULL) {
        return DS_INVALID;
    }
    if (stack->count == 0) {
        return DS_EMPTY;
    }
    *out = stack->top->value;
    return DS_OK;
}
```

```c
        lqueue_push(&linked, 10) != DS_OK || lqueue_push(&linked, 20) != DS_OK;
    int a = 0, b = 0;
    if (!failed) {
        failed = queue_pop(&ring, &a) != DS_OK || lqueue_pop(&linked, &b) != DS_OK;
    }
    if (!failed) {
        printf("first=%d,%d\n", a, b);
    }
    if (!failed) failed = queue_push(&ring, 30) != DS_OK || queue_push(&ring, 40) != DS_OK ||
        lqueue_push(&linked, 30) != DS_OK || lqueue_push(&linked, 40) != DS_OK;
    for (int i = 0; i < 3 && !failed; i++) {
        failed = queue_pop(&ring, &a) != DS_OK || lqueue_pop(&linked, &b) != DS_OK;
        if (!failed) {
            printf("next=%d,%d\n", a, b);
        }
    }
    if (!failed) {
        printf("linked_empty=%d\n", linked.head == NULL && linked.tail == NULL);
    }
    lqueue_clear(&linked);
    return failed;
}
```

```c
#include "queue.h"
#include <stdlib.h>

DsStatus queue_init(Queue *queue, size_t capacity)
{
    if (capacity == 0 || capacity > QUEUE_MAX_CAPACITY) {
        return DS_INVALID;
    }
    queue->capacity = capacity;
    queue->head = queue->count = 0;
    return DS_OK;
}
DsStatus queue_push(Queue *queue, int value)
{
    if (queue->count == queue->capacity) {
        return DS_FULL;
    }
    size_t tail = (queue->head + queue->count) % queue->capacity;
    queue->data[tail] = value;
    queue->count++;
    return DS_OK;
}
DsStatus queue_pop(Queue *queue, int *out)
{
    if (out == NULL) {
        return DS_INVALID;
    }
    if (queue->count == 0) {
        return DS_EMPTY;
    }
    *out = queue->data[queue->head];
    queue->head = (queue->head + 1) % queue->capacity;
    queue->count--;
    return DS_OK;
}
DsStatus queue_peek(const Queue *queue, int *out)
{
    if (out == NULL) {
        return DS_INVALID;
    }
    if (queue->count == 0) {
        return DS_EMPTY;
    }
    *out = queue->data[queue->head];
    return DS_OK;
}
void lqueue_init(LinkedQueue *queue) {
    queue->head = queue->tail = NULL;
    queue->count = 0;
}
void lqueue_clear(LinkedQueue *queue)
{
    QNode *node = queue->head;
    while (node != NULL) {
        QNode *next = node->next;
        free(node);
        node = next;
    }
    lqueue_init(queue);
}
DsStatus lqueue_push(LinkedQueue *queue, int value)
{
    if (queue->count == DS_MAX_ITEMS) {
        return DS_FULL;
    }
    QNode *node = malloc(sizeof *node);
    if (node == NULL) {
        return DS_MEMORY;
    }
    node->value = value;
    node->next = NULL;
    if (queue->tail == NULL) {
        queue->head = node;
    }
    else {
        queue->tail->next = node;
    }
    queue->tail = node;
    queue->count++;
    return DS_OK;
}
DsStatus lqueue_pop(LinkedQueue *queue, int *out)
{
    if (out == NULL) {
        return DS_INVALID;
    }
    if (queue->count == 0) {
        return DS_EMPTY;
    }
    QNode *node = queue->head;
    int saved = node->value;
    queue->head = node->next;
    queue->count--;
    if (queue->head == NULL) {
        queue->tail = NULL;
    }
    free(node);
    *out = saved;
    return DS_OK;
}
DsStatus lqueue_peek(const LinkedQueue *queue, int *out)
{
    if (out == NULL) {
        return DS_INVALID;
    }
    if (queue->count == 0) {
        return DS_EMPTY;
    }
    *out = queue->head->value;
    return DS_OK;
}
```

```c
#include "expression.h"
#include "stack.h"
#include "queue.h"
#include <string.h>

ExprStatus expression_evaluate(const char *text, int *out, size_t *error_token)
{
    if (text == NULL || out == NULL || error_token == NULL) {
        return E_INVALID;
    }
    if (strlen(text) > 127) {
        *error_token = 0;
        return E_LENGTH;
    }
    Queue tokens;
    if (queue_init(&tokens, QUEUE_MAX_CAPACITY) != DS_OK) {
        return E_INVALID;
    }
    size_t number = 0;
    for (size_t i = 0; text[i] != '\0'; i++) {
        char ch = text[i];
        if (ch == ' ') {
            continue;
        }
        if (!((ch >= '0' && ch <= '9') || ch == '+' || ch == '-' || ch == '*' || ch == '/') ||
            (text[i + 1] != '\0' && text[i + 1] != ' ')) {
            *error_token = number;
            return E_TOKEN;
        }
        if (queue_push(&tokens, ch) != DS_OK) {
            *error_token = number;
            return E_CAPACITY;
        }
        number++;
    }
    ArrayStack values;
    stack_init(&values);
    size_t index = 0;
    int token = 0;
    while (queue_pop(&tokens, &token) == DS_OK) {
        if (token >= '0' && token <= '9') {
            if (stack_push(&values, token - '0') != DS_OK) {
                *error_token = index;
                return E_CAPACITY;
            }
        } else {
            int right = 0, left = 0;
            if (stack_pop(&values, &right) != DS_OK || stack_pop(&values, &left) != DS_OK) {
                *error_token = index;
                return E_ARITY;
            }
            long result = 0;
            switch (token) {
            case '+': result = (long)left + right; break;
            case '-': result = (long)left - right; break;
            case '*': result = (long)left * right; break;
            case '/':
                if (right == 0) {
                    *error_token = index;
                    return E_DIVZERO;
                }
                result = (long)left / right;
                break;
            default: return E_INVALID;
            }
            if (result < -9999 || result > 9999) {
                *error_token = index;
                return E_RANGE;
            }
            if (stack_push(&values, (int)result) != DS_OK) {
                *error_token = index;
                return E_CAPACITY;
            }
        }
        index++;
    }
    int final = 0;
    if (stack_pop(&values, &final) != DS_OK || values.count != 0) {
        *error_token = number;
        return E_ARITY;
    }
    *out = final;
    return E_OK;
}

const char *expression_status_name(ExprStatus status)
{
    switch (status) {
    case E_OK: return "ok";
    case E_TOKEN: return "token";
    case E_CAPACITY: return "capacity";
    case E_ARITY: return "arity";
    case E_DIVZERO: return "division-by-zero";
    case E_RANGE: return "range";
    case E_LENGTH: return "length";
    default: return "invalid";
    }
}
```
