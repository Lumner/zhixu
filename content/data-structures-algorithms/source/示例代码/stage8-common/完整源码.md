#include "graph_internal.h"
#include <limits.h>
GStatus graph_init(Graph *graph, size_t n, int directed)
{
    if (graph == NULL || (directed != 0 && directed != 1)) { return G_INVALID; }
    if (n > GRAPH_MAX) { return G_RANGE; }
    Graph next = {0}; next.n = n; next.directed = directed; *graph = next; return G_OK;
}
GStatus graph_add(Graph *graph, size_t from, size_t to, int weight)
{
    if (graph == NULL || from >= graph->n || to >= graph->n) { return G_INVALID; }
    if (weight < -GRAPH_WEIGHT_MAX || weight > GRAPH_WEIGHT_MAX) { return G_RANGE; }
    if (graph->has[from][to]) { return G_DUPLICATE; }
    graph->has[from][to] = 1; graph->weight[from][to] = weight;
    if (!graph->directed) { graph->has[to][from] = 1; graph->weight[to][from] = weight; }
    graph->edges++; return G_OK;
}
GStatus graph_remove(Graph *graph, size_t from, size_t to)
{
    if (graph == NULL || from >= graph->n || to >= graph->n) { return G_INVALID; }
    if (!graph->has[from][to]) { return G_MISSING; }
    graph->has[from][to] = 0; graph->weight[from][to] = 0;
    if (!graph->directed) { graph->has[to][from] = 0; graph->weight[to][from] = 0; }
    graph->edges--; return G_OK;
}
GStatus graph_adjacency(const Graph *graph, Adjacency *out)
{
    if (graph == NULL || out == NULL) { return G_INVALID; }
    Adjacency next = {0}; next.n = graph->n;
    for (size_t u = 0; u < GRAPH_MAX; u++) { next.head[u] = GRAPH_ARCS; }
    for (size_t u = 0; u < graph->n; u++) {
        for (size_t end = graph->n; end > 0; end--) {
            size_t v = end - 1;
            if (graph->has[u][v]) {
                size_t index = next.count++;
                next.arcs[index] = (Arc){v,graph->weight[u][v],next.head[u]}; next.head[u] = index;
            }
        }
    }
    *out = next; return G_OK;
}
GStatus graph_cost_sum(long left, long right, long *out)
{
    if (out == NULL) { return G_INVALID; }
    if ((right > 0 && left > LONG_MAX - right) || (right < 0 && left < LONG_MIN - right)) { return G_RANGE; }
    *out = left + right; return G_OK;
}

完整文件：[walk.c](../44-traversal/walk.c)。

#include "walk.h"
static void prepare(Walk *walk, size_t n, size_t source)
{
    *walk = (Walk){0}; walk->n = n; walk->source = source;
    for (size_t i = 0; i < GRAPH_MAX; i++) { walk->depth[i] = walk->parent[i] = GRAPH_MAX; }
    walk->depth[source] = 0;
}
static void visit(const Graph *graph, size_t u, Walk *walk)
{
    walk->order[walk->count++] = u;
    for (size_t v = 0; v < graph->n; v++) {
        if (graph->has[u][v] && walk->depth[v] == GRAPH_MAX) {
            walk->parent[v] = u; walk->depth[v] = walk->depth[u] + 1; visit(graph, v, walk);
        }
    }
}
GStatus graph_dfs(const Graph *graph, size_t source, Walk *out)
{
    if (graph == NULL || out == NULL || source >= graph->n) { return G_INVALID; }
    Walk next; prepare(&next, graph->n, source); visit(graph, source, &next); *out = next; return G_OK;
}
GStatus graph_bfs(const Graph *graph, size_t source, Walk *out)
{
    if (graph == NULL || out == NULL || source >= graph->n) { return G_INVALID; }
    Walk next; prepare(&next, graph->n, source); size_t queue[GRAPH_MAX], head = 0, tail = 0;
    queue[tail++] = source;
    while (head < tail) {
        size_t u = queue[head++]; next.order[next.count++] = u;
        for (size_t v = 0; v < graph->n; v++) {
            if (graph->has[u][v] && next.depth[v] == GRAPH_MAX) {
                next.depth[v] = next.depth[u] + 1; next.parent[v] = u; queue[tail++] = v;
            }
        }
    }
    *out = next; return G_OK;
}
GStatus graph_components(const Graph *graph, Components *out)
{
    if (graph == NULL || out == NULL || graph->directed) { return G_INVALID; }
    Components next = {0}; next.n = graph->n;
    for (size_t i = 0; i < GRAPH_MAX; i++) { next.label[i] = GRAPH_MAX; }
    for (size_t u = 0; u < graph->n; u++) {
        if (next.label[u] != GRAPH_MAX) { continue; }
        Walk walk; (void)graph_bfs(graph, u, &walk);
        for (size_t i = 0; i < walk.count; i++) { next.label[walk.order[i]] = next.count; }
        next.count++;
    }
    *out = next; return G_OK;
}

#include "topological.h"
GStatus graph_topological(const Graph *graph, Topological *out)
{
    if (graph == NULL || out == NULL || !graph->directed) { return G_INVALID; }
    size_t indegree[GRAPH_MAX] = {0}, queue[GRAPH_MAX], head = 0, tail = 0; Topological next = {0};
    for (size_t u = 0; u < graph->n; u++) { for (size_t v = 0; v < graph->n; v++) { if (graph->has[u][v]) { indegree[v]++; } } }
    for (size_t u = 0; u < graph->n; u++) { if (indegree[u] == 0) { queue[tail++] = u; } }
    while (head < tail) {
        size_t u = queue[head++]; next.vertices[next.count++] = u;
        for (size_t v = 0; v < graph->n; v++) { if (graph->has[u][v] && --indegree[v] == 0) { queue[tail++] = v; } }
    }
    if (next.count != graph->n) { return G_CYCLE; }
    *out = next; return G_OK;
}

GStatus graph_dijkstra(const Graph *graph, size_t source, Shortest *out)
{
    if (graph == NULL || out == NULL || source >= graph->n) { return G_INVALID; }
    for (size_t u = 0; u < graph->n; u++) {
        for (size_t v = 0; v < graph->n; v++) { if (graph->has[u][v] && graph->weight[u][v] < 0) { return G_NEGATIVE; } }
    }
    Shortest next; graph_paths_begin(&next, graph->n, source); int done[GRAPH_MAX] = {0};
    for (size_t step = 0; step < graph->n; step++) {
        size_t u = GRAPH_MAX;
        for (size_t v = 0; v < graph->n; v++) {
            if (!done[v] && next.state[v] == PATH_FINITE && (u == GRAPH_MAX || next.distance[v] < next.distance[u])) { u = v; }
        }
        if (u == GRAPH_MAX) { break; }
        done[u] = 1;
        for (size_t v = 0; v < graph->n; v++) {
            if (!graph->has[u][v] || done[v]) { continue; }
            long cost; GStatus status = graph_cost_sum(next.distance[u], graph->weight[u][v], &cost);
            if (status != G_OK) { return status; }
            if (next.state[v] == PATH_UNREACHABLE || cost < next.distance[v]) {
                next.state[v] = PATH_FINITE; next.distance[v] = cost; next.parent[v] = u;
            }
        }
    }
    *out = next; return G_OK;
}

#include "shortest_internal.h"
void graph_paths_begin(Shortest *paths, size_t n, size_t source)
{
    *paths = (Shortest){0}; paths->n = n; paths->source = source;
    for (size_t i = 0; i < GRAPH_MAX; i++) { paths->parent[i] = GRAPH_MAX; }
    paths->state[source] = PATH_FINITE;
}
const char *path_state_name(PathState state)
{
    switch (state) { case PATH_UNREACHABLE: return "unreachable"; case PATH_FINITE: return "finite"; case PATH_NEGATIVE: return "negative"; default: return "invalid"; }
}
GStatus graph_path(const Shortest *paths, size_t target, GraphPath *out)
{
    if (paths == NULL || out == NULL || target >= paths->n) { return G_INVALID; }
    if (paths->state[target] == PATH_UNREACHABLE) { return G_MISSING; }
    if (paths->state[target] == PATH_NEGATIVE) { return G_NEGATIVE; }
    GraphPath next = {0}; size_t reverse[GRAPH_MAX], n = 0, current = target;
    for (;;) {
        if (current >= paths->n || n == paths->n) { return G_INVALID; }
        reverse[n++] = current;
        if (current == paths->source) { break; }
        current = paths->parent[current];
    }
    next.count = n; next.cost = paths->distance[target];
    for (size_t i = 0; i < n; i++) { next.vertices[i] = reverse[n - 1 - i]; }
    *out = next; return G_OK;
}

#include "shortest_internal.h"
GStatus graph_bellman(const Graph *graph, size_t source, Shortest *out)
{
    if (graph == NULL || out == NULL || source >= graph->n) { return G_INVALID; }
    Shortest next; graph_paths_begin(&next, graph->n, source);
    for (size_t pass = 1; pass < graph->n; pass++) {
        Shortest previous = next; int changed = 0;
        for (size_t u = 0; u < graph->n; u++) {
            if (previous.state[u] != PATH_FINITE) { continue; }
            for (size_t v = 0; v < graph->n; v++) {
                if (!graph->has[u][v]) { continue; }
                long cost; GStatus status = graph_cost_sum(previous.distance[u], graph->weight[u][v], &cost);
                if (status != G_OK) { return status; }
                if (next.state[v] == PATH_UNREACHABLE || cost < next.distance[v]) {
                    next.state[v] = PATH_FINITE; next.distance[v] = cost; next.parent[v] = u; changed = 1;
                }
            }
        }
        if (!changed) { break; }
    }
    int affected[GRAPH_MAX] = {0};
    for (size_t u = 0; u < graph->n; u++) {
        if (next.state[u] != PATH_FINITE) { continue; }
        for (size_t v = 0; v < graph->n; v++) {
            if (!graph->has[u][v]) { continue; }
            long cost; GStatus status = graph_cost_sum(next.distance[u], graph->weight[u][v], &cost);
            if (status != G_OK) { return status; }
            if (next.state[v] == PATH_UNREACHABLE || cost < next.distance[v]) { affected[v] = 1; }
        }
    }
    for (size_t pass = 0; pass < graph->n; pass++) {
        for (size_t u = 0; u < graph->n; u++) { if (affected[u]) { for (size_t v = 0; v < graph->n; v++) { if (graph->has[u][v]) { affected[v] = 1; } } } }
    }
    for (size_t v = 0; v < graph->n; v++) { if (affected[v]) { next.state[v] = PATH_NEGATIVE; next.parent[v] = GRAPH_MAX; } }
    *out = next; return G_OK;
}

#include "shortest_internal.h"
GStatus graph_floyd(const Graph *graph, AllPairs *out)
{
    if (graph == NULL || out == NULL) { return G_INVALID; }
    AllPairs next = {0}; next.n = graph->n;
    for (size_t u = 0; u < graph->n; u++) {
        next.state[u][u] = PATH_FINITE;
        for (size_t v = 0; v < graph->n; v++) {
            if (graph->has[u][v] && (u != v || graph->weight[u][v] < 0)) { next.state[u][v] = PATH_FINITE; next.distance[u][v] = graph->weight[u][v]; }
        }
    }
    for (size_t k = 0; k < graph->n; k++) {
        AllPairs previous = next;
        for (size_t u = 0; u < graph->n; u++) {
            for (size_t v = 0; v < graph->n; v++) {
                if (previous.state[u][k] != PATH_FINITE || previous.state[k][v] != PATH_FINITE) { continue; }
                long cost; GStatus status = graph_cost_sum(previous.distance[u][k], previous.distance[k][v], &cost);
                if (status != G_OK) { return status; }
                if (next.state[u][v] == PATH_UNREACHABLE || cost < next.distance[u][v]) { next.state[u][v] = PATH_FINITE; next.distance[u][v] = cost; }
            }
        }
    }
    AllPairs finite = next;
    for (size_t u = 0; u < graph->n; u++) {
        for (size_t v = 0; v < graph->n; v++) {
            for (size_t k = 0; k < graph->n; k++) {
                if (finite.state[u][k] == PATH_FINITE && finite.state[k][v] == PATH_FINITE && finite.distance[k][k] < 0) { next.state[u][v] = PATH_NEGATIVE; }
            }
        }
    }
    *out = next; return G_OK;
}

#include "dsu.h"
GStatus dsu_init(DisjointSet *sets, size_t n)
{
    if (sets == NULL) { return G_INVALID; }
    if (n > GRAPH_MAX) { return G_RANGE; }
    DisjointSet next = {0}; next.n = next.components = n;
    for (size_t i = 0; i < n; i++) { next.parent[i] = i; next.size[i] = 1; }
    *sets = next; return G_OK;
}
GStatus dsu_find(DisjointSet *sets, size_t value, size_t *out)
{
    if (sets == NULL || out == NULL || value >= sets->n) { return G_INVALID; }
    size_t root = value;
    while (sets->parent[root] != root) { root = sets->parent[root]; }
    while (value != root) { size_t next = sets->parent[value]; sets->parent[value] = root; value = next; }
    *out = root; return G_OK;
}
GStatus dsu_union(DisjointSet *sets, size_t a, size_t b, int *merged)
{
    if (sets == NULL || merged == NULL || a >= sets->n || b >= sets->n) { return G_INVALID; }
    size_t ra, rb; (void)dsu_find(sets, a, &ra); (void)dsu_find(sets, b, &rb);
    *merged = 0;
    if (ra != rb) {
        if (sets->size[ra] < sets->size[rb] || (sets->size[ra] == sets->size[rb] && ra > rb)) { size_t saved = ra; ra = rb; rb = saved; }
        sets->parent[rb] = ra; sets->size[ra] += sets->size[rb]; sets->components--; *merged = 1;
    }
    return G_OK;
}

static int compare(Edge a, Edge b)
{
    if (a.weight != b.weight) { return a.weight < b.weight ? -1 : 1; }
    if (a.from != b.from) { return a.from < b.from ? -1 : 1; }
    return a.to < b.to ? -1 : a.to > b.to ? 1 : 0;
}
static void swap(Edge *a, Edge *b) { Edge saved = *a; *a = *b; *b = saved; }
static void edge_down(Edge *edges, size_t root, size_t n)
{
    while (root < n / 2) {
        size_t child = root * 2 + 1;
        if (child + 1 < n && compare(edges[child + 1], edges[child]) > 0) { child++; }
        if (compare(edges[root], edges[child]) >= 0) { break; }
        swap(&edges[root], &edges[child]); root = child;
    }
}
static void edge_sort(Edge *edges, size_t n)
{
    for (size_t i = n / 2; i > 0; i--) { edge_down(edges, i - 1, n); }
    for (size_t end = n; end > 1; end--) { swap(&edges[0], &edges[end - 1]); edge_down(edges, 0, end - 1); }
}
GStatus graph_kruskal(const Graph *graph, Forest *out)
{
    if (graph == NULL || out == NULL || graph->directed) { return G_INVALID; }
    Edge edges[GRAPH_ARCS]; size_t n = 0;
    for (size_t u = 0; u < graph->n; u++) { for (size_t v = u + 1; v < graph->n; v++) { if (graph->has[u][v]) { edges[n++] = (Edge){u,v,graph->weight[u][v]}; } } }
    edge_sort(edges, n); DisjointSet sets; (void)dsu_init(&sets, graph->n);
    Forest next = {0}; next.vertices = graph->n;
    for (size_t i = 0; i < n; i++) {
        int merged; (void)dsu_union(&sets, edges[i].from, edges[i].to, &merged);
        if (merged) {
            GStatus status = graph_cost_sum(next.total, edges[i].weight, &next.total);
            if (status != G_OK) { return status; }
            next.edges[next.count++] = edges[i];
        }
    }
    next.components = sets.components; *out = next; return G_OK;
}
GStatus graph_prim(const Graph *graph, Forest *out)
{
    if (graph == NULL || out == NULL || graph->directed) { return G_INVALID; }
    Forest next = {0}; next.vertices = graph->n;
    int used[GRAPH_MAX] = {0}, known[GRAPH_MAX] = {0}, best[GRAPH_MAX] = {0}; size_t parent[GRAPH_MAX];
    for (size_t i = 0; i < GRAPH_MAX; i++) { parent[i] = GRAPH_MAX; }
    for (size_t root = 0; root < graph->n; root++) {
        if (used[root]) { continue; }
        next.components++; known[root] = 1; best[root] = 0;
        for (;;) {
            size_t u = GRAPH_MAX;
            for (size_t v = 0; v < graph->n; v++) { if (!used[v] && known[v] && (u == GRAPH_MAX || best[v] < best[u])) { u = v; } }
            if (u == GRAPH_MAX) { break; }
            used[u] = 1;
            if (parent[u] != GRAPH_MAX) {
                size_t a = parent[u] < u ? parent[u] : u, b = parent[u] < u ? u : parent[u];
                next.edges[next.count++] = (Edge){a,b,best[u]};
                GStatus status = graph_cost_sum(next.total, best[u], &next.total); if (status != G_OK) { return status; }
            }
            for (size_t v = 0; v < graph->n; v++) {
                if (!used[v] && graph->has[u][v] && (!known[v] || graph->weight[u][v] < best[v] || (graph->weight[u][v] == best[v] && u < parent[v]))) {
                    known[v] = 1; best[v] = graph->weight[u][v]; parent[v] = u;
                }
            }
        }
    }
    *out = next; return G_OK;
}
