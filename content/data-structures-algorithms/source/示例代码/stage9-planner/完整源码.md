#include "planner_internal.h"
#include <string.h>
void planner_init(Planner *app)
{
    memset(app, 0, sizeof(*app));
    indexed_init(&app->catalog);
    for (size_t i = 0; i < RECORD_LIMIT; i++) { app->membership[i] = P_CLASSES; }
    (void)graph_init(&app->study, 0, 1);
    (void)graph_init(&app->roads, 0, 0);
}
void planner_clear(Planner *app)
{
    indexed_clear(&app->catalog);
    planner_init(app);
}
static void swap_owners(Planner *left, Planner *right)
{
    Planner temporary = *left;
    *left = *right;
    *right = temporary;
}
PStatus planner_number(const char *text, int limit, int *out)
{
    if (text == NULL || out == NULL || limit < 0) { return P_INVALID; }
    if (*text == '\0') { return P_FORMAT; }
    int value = 0;
    for (size_t i = 0; text[i] != '\0'; i++) {
        if (text[i] < '0' || text[i] > '9') { return P_FORMAT; }
        int digit = text[i] - '0';
        if (digit > limit || value > (limit - digit) / 10) { return P_LIMIT; }
        value = value * 10 + digit;
    }
    *out = value;
    return P_OK;
}
PStatus p_catalog_status(CatStatus status)
{
    switch (status) {
    case CAT_OK: return P_OK;
    case CAT_DUPLICATE: return P_DUPLICATE;
    case CAT_LIMIT: return P_LIMIT;
    case CAT_NOT_FOUND: return P_MISSING;
    case CAT_MEMORY: return P_MEMORY;
    case CAT_IO: return P_IO;
    default: return P_FORMAT;
    }
}
PStatus p_graph_status(GStatus status)
{
    switch (status) {
    case G_OK: return P_OK;
    case G_DUPLICATE: return P_DUPLICATE;
    case G_RANGE: case G_SPACE: return P_LIMIT;
    case G_MISSING: return P_MISSING;
    case G_CYCLE: return P_CYCLE;
    default: return P_INVALID;
    }
}
PStatus planner_load(FILE *catalog, FILE *relations, Planner *out)
{
    if (catalog == NULL || relations == NULL || out == NULL) { return P_INVALID; }
    Planner candidate;
    planner_init(&candidate);
    PStatus status = p_catalog_status(v3_load(catalog, &candidate.catalog));
    if (status == P_OK) { status = p_relations(relations, &candidate); }
    if (status == P_OK) { swap_owners(out, &candidate); }
    planner_clear(&candidate);
    return status;
}
PStatus planner_load_files(const char *catalog, const char *relations, Planner *out)
{
    if (catalog == NULL || relations == NULL || out == NULL) { return P_INVALID; }
    FILE *records = fopen(catalog, "rb");
    if (records == NULL) { return P_IO; }
    FILE *links = fopen(relations, "rb");
    if (links == NULL) { (void)fclose(records); return P_IO; }
    Planner candidate;
    planner_init(&candidate);
    PStatus status = planner_load(records, links, &candidate);
    int close_records = fclose(records);
    int close_links = fclose(links);
    if (status == P_OK && (close_records != 0 || close_links != 0)) { status = P_IO; }
    if (status == P_OK) { swap_owners(out, &candidate); }
    planner_clear(&candidate);
    return status;
}
PStatus planner_set_minutes(Planner *app, int id, int minutes)
{
    if (app == NULL) { return P_INVALID; }
    if (id < 0 || id > 9999 || minutes < 0 || minutes > 1440) { return P_LIMIT; }
    IRecord record;
    if (indexed_find(&app->catalog, id, &record) != H_OK) { return P_MISSING; }
    return p_catalog_status(v3_hash_status(indexed_set_minutes(&app->catalog, id, minutes)));
}
PStatus planner_export(FILE *out, const Planner *app)
{
    if (out == NULL || app == NULL) { return P_INVALID; }
    return p_catalog_status(v3_save(out, &app->catalog));
}
const char *planner_status_name(PStatus status)
{
    switch (status) {
    case P_OK: return "ok";
    case P_INVALID: return "invalid argument";
    case P_FORMAT: return "bad format";
    case P_DUPLICATE: return "duplicate item";
    case P_LIMIT: return "limit exceeded";
    case P_MISSING: return "missing reference or item";
    case P_CYCLE: return "cycle";
    case P_MEMORY: return "out of memory";
    case P_IO: return "I/O error";
    default: return "unknown error";
    }
}

#include "planner.h"
#include <string.h>
static void record_print(const IRecord *row)
{
    printf("id=%d minutes=%d title=%s\n", row->id, row->minutes, row->title);
}
static void ids_print(const char *name, const PIds *ids)
{
    printf("%s=", name);
    if (ids->count == 0) { fputs("empty", stdout); }
    for (size_t i = 0; i < ids->count; i++) { printf("%s%d", i == 0 ? "" : ",", ids->ids[i]); }
    puts("");
}
static int valid_command(int argc, char **argv)
{
    if (argc < 4) { return 0; }
    const char *command = argv[3];
    if (!strcmp(command,"check") || !strcmp(command,"list") || !strcmp(command,"categories") ||
        !strcmp(command,"order") || !strcmp(command,"export")) { return argc == 4; }
    if (!strcmp(command,"find") || !strcmp(command,"title")) { return argc == 5; }
    if (!strcmp(command,"route") || !strcmp(command,"minutes")) { return argc == 6; }
    if (!strcmp(command,"ready")) { return argc <= 4 + GRAPH_MAX; }
    return 0;
}
static PStatus execute(Planner *app, int argc, char **argv)
{
    const char *command = argv[3];
    if (!strcmp(command,"check")) {
        printf("records=%zu categories=%zu study=%zu places=%zu\n", app->catalog.count,
               app->category_count, app->study_count, app->place_count);
        return P_OK;
    }
    if (!strcmp(command,"export")) { return planner_export(stdout,app); }
    if (!strcmp(command,"find") || !strcmp(command,"minutes") || !strcmp(command,"route")) {
        int first, second;
        PStatus status = planner_number(argv[4],9999,&first);
        if (status != P_OK) { return status; }
        if (!strcmp(command,"find")) {
            IRecord row;
            status = planner_find(app,first,&row);
            if (status == P_OK) { record_print(&row); }
            return status;
        }
        status = planner_number(argv[5],!strcmp(command,"minutes") ? 1440 : 9999,&second);
        if (status != P_OK) { return status; }
        if (!strcmp(command,"minutes")) {
            status = planner_set_minutes(app,first,second);
            return status == P_OK ? planner_export(stdout,app) : status;
        }
        PRoute route;
        status = planner_route(app,first,second,&route);
        if (status != P_OK) { return status; }
        if (!route.reachable) { puts("unreachable"); return P_OK; }
        printf("cost=%ld path=",route.cost);
        for (size_t i = 0; i < route.count; i++) { printf("%s%d",i == 0 ? "" : ",",route.ids[i]); }
        puts(""); return P_OK;
    }
    if (!strcmp(command,"list") || !strcmp(command,"title")) {
        PRecords rows;
        PStatus status = !strcmp(command,"list") ? planner_view(app,&rows) : planner_title(app,argv[4],&rows);
        if (status != P_OK) { return status; }
        if (rows.count == 0) { puts("no matches"); }
        for (size_t i = 0; i < rows.count; i++) { record_print(&rows.records[i]); }
        return P_OK;
    }
    if (!strcmp(command,"categories")) {
        PCategoryView view;
        PStatus status = planner_categories(app,&view);
        if (status != P_OK) { return status; }
        if (view.count == 0) { puts("empty categories"); }
        for (size_t i = 0; i < view.count; i++) {
            size_t category = view.positions[i];
            for (size_t d = 0; d < view.depths[i]; d++) { fputs("  ",stdout); }
            printf("category=%d label=%s\n",app->categories[category].id,app->categories[category].label);
            for (size_t j = 0; j < app->catalog.count; j++) {
                if (app->membership[j] == category) {
                    for (size_t d = 0; d <= view.depths[i]; d++) { fputs("  ",stdout); }
                    record_print(&app->catalog.records[j]);
                }
            }
        }
        return P_OK;
    }
    PIds ids;
    PStatus status;
    if (!strcmp(command,"order")) {
        status = planner_order(app,&ids);
    } else {
        int done[GRAPH_MAX]; size_t n = (size_t)(argc - 4);
        for (size_t i = 0; i < n; i++) {
            status = planner_number(argv[i + 4],9999,&done[i]);
            if (status != P_OK) { return status; }
        }
        status = planner_ready(app,done,n,&ids);
    }
    if (status == P_OK) { ids_print(command,&ids); }
    return status;
}
int main(int argc, char **argv)
{
    if (!valid_command(argc,argv)) {
        fputs("usage: planner catalog relations check|list|find ID|title PATTERN|categories|order|ready [ID...]|route FROM TO|minutes ID VALUE|export\n",stderr);
        return 1;
    }
    Planner app;
    planner_init(&app);
    PStatus status = planner_load_files(argv[1],argv[2],&app);
    if (status == P_OK) { status = execute(&app,argc,argv); }
    if (status == P_OK && (fflush(stdout) != 0 || ferror(stdout))) { status = P_IO; }
    planner_clear(&app);
    if (status != P_OK) { fprintf(stderr,"planner: %s\n",planner_status_name(status)); }
    return status == P_OK ? 0 : 1;
}

#include "../27-matching/matching.h"

PStatus planner_find(const Planner *app, int id, IRecord *out)
{
    if (app == NULL || out == NULL) { return P_INVALID; }
    if (id < 0 || id > 9999) { return P_LIMIT; }
    return p_catalog_status(v3_hash_status(indexed_find(&app->catalog, id, out)));
}
PStatus planner_view(const Planner *app, PRecords *out)
{
    if (app == NULL || out == NULL) { return P_INVALID; }
    PRecords next = {0};
    SortStatus status = catalog_view(&app->catalog, next.records, RECORD_LIMIT, &next.stats);
    if (status != S_OK) { return status == S_MEMORY ? P_MEMORY : P_INVALID; }
    next.count = app->catalog.count;
    *out = next;
    return P_OK;
}
PStatus planner_title(const Planner *app, const char *pattern, PRecords *out)
{
    if (app == NULL || pattern == NULL || out == NULL) { return P_INVALID; }
    size_t length = strlen(pattern);
    if (length > MATCH_PATTERN_MAX) { return P_LIMIT; }
    PRecords next = {0};
    for (size_t i = 0; i < app->catalog.count; i++) {
        const IRecord *row = &app->catalog.records[i];
        Match match;
        if (kmp_match(row->title, strlen(row->title), pattern, length, &match) != M_OK) { return P_INVALID; }
        if (match.found) { next.records[next.count++] = *row; }
    }
    *out = next;
    return P_OK;
}
static void category_visit(const Planner *app, size_t position, size_t depth, PCategoryView *out)
{
    out->positions[out->count] = position;
    out->depths[out->count++] = depth;
    for (size_t i = 0; i < app->category_count; i++) {
        if (app->categories[i].parent_id == app->categories[position].id) {
            category_visit(app, i, depth + 1, out);
        }
    }
}
PStatus planner_categories(const Planner *app, PCategoryView *out)
{
    if (app == NULL || out == NULL) { return P_INVALID; }
    PCategoryView next = {0};
    for (size_t i = 0; i < app->category_count; i++) {
        if (app->categories[i].parent_id == -1) { category_visit(app, i, 0, &next); }
    }
    *out = next;
    return P_OK;
}
PStatus planner_order(const Planner *app, PIds *out)
{
    if (app == NULL || out == NULL) { return P_INVALID; }
    Topological order;
    PStatus status = p_graph_status(graph_topological(&app->study, &order));
    if (status != P_OK) { return status; }
    PIds next = {0};
    next.count = order.count;
    for (size_t i = 0; i < order.count; i++) { next.ids[i] = app->study_ids[order.vertices[i]]; }
    *out = next;
    return P_OK;
}
PStatus planner_ready(const Planner *app, const int *done, size_t n, PIds *out)
{
    if (app == NULL || out == NULL || (n != 0 && done == NULL)) { return P_INVALID; }
    if (n > GRAPH_MAX) { return P_LIMIT; }
    unsigned char finished[GRAPH_MAX] = {0};
    for (size_t i = 0; i < n; i++) {
        size_t position = p_study(app, done[i]);
        if (position == GRAPH_MAX) { return P_MISSING; }
        if (finished[position]) { return P_DUPLICATE; }
        finished[position] = 1;
    }
    for (size_t u = 0; u < app->study_count; u++) {
        for (size_t v = 0; v < app->study_count; v++) {
            if (app->study.has[u][v] && finished[v] && !finished[u]) { return P_INVALID; }
        }
    }
    MinHeap queue;
    heap_init(&queue);
    for (size_t v = 0; v < app->study_count; v++) {
        int blocked = finished[v];
        for (size_t u = 0; u < app->study_count; u++) {
            if (app->study.has[u][v] && !finished[u]) { blocked = 1; }
        }
        if (!blocked) {
            IRecord row;
            if (planner_find(app, app->study_ids[v], &row) != P_OK) { return P_MISSING; }
            if (heap_push(&queue, (Job){row.minutes,row.id}) != T_OK) { return P_INVALID; }
        }
    }
    PIds next = {0};
    while (queue.count != 0) {
        Job job;
        if (heap_pop(&queue, &job) != T_OK) { return P_INVALID; }
        next.ids[next.count++] = job.id;
    }
    *out = next;
    return P_OK;
}
PStatus planner_route(const Planner *app, int source, int target, PRoute *out)
{
    if (app == NULL || out == NULL) { return P_INVALID; }
    size_t from = p_place(app, source), to = p_place(app, target);
    if (from == GRAPH_MAX || to == GRAPH_MAX) { return P_MISSING; }
    Shortest paths;
    PStatus status = p_graph_status(graph_dijkstra(&app->roads, from, &paths));
    if (status != P_OK) { return status; }
    PRoute next = {0};
    if (paths.state[to] == PATH_FINITE) {
        GraphPath path;
        status = p_graph_status(graph_path(&paths, to, &path));
        if (status != P_OK) { return status; }
        next.reachable = 1; next.cost = path.cost; next.count = path.count;
        for (size_t i = 0; i < path.count; i++) { next.ids[i] = app->places[path.vertices[i]].id; }
    }
    *out = next;
    return P_OK;
}

static PStatus line_read(FILE *in, char *line, int *end)
{
    size_t n = 0;
    int ch, too_long = 0;
    *end = 0;
    while ((ch = fgetc(in)) != EOF && ch != '\n') {
        if (n + 1 < P_LINE) { line[n++] = (char)ch; }
        else { too_long = 1; }
    }
    if (ferror(in)) { return P_IO; }
    if (ch == EOF && n == 0 && !too_long) { *end = 1; return P_OK; }
    if (too_long) { return P_LIMIT; }
    if (n != 0 && line[n - 1] == '\r' && ch == '\n') { n--; }
    for (size_t i = 0; i < n; i++) {
        unsigned char byte = (unsigned char)line[i];
        if (byte < 32 || byte == 127) { return P_FORMAT; }
    }
    line[n] = '\0';
    return P_OK;
}
static PStatus fields(char *line, char **parts, size_t *count)
{
    size_t n = 1;
    parts[0] = line;
    for (size_t i = 0; line[i] != '\0'; i++) {
        if (line[i] == '|') {
            if (n == 4) { return P_FORMAT; }
            line[i] = '\0'; parts[n++] = line + i + 1;
        }
    }
    *count = n;
    return P_OK;
}
static PStatus label_copy(char *out, const char *text)
{
    size_t n = strlen(text);
    if (n == 0) { return P_FORMAT; }
    if (n >= P_LABEL) { return P_LIMIT; }
    memcpy(out, text, n + 1);
    return P_OK;
}
static PStatus append(Planner *app, Pending *pending, int type, char **parts)
{
    int first, second = 0, third = 0;
    PStatus status = planner_number(parts[1], 9999, &first);
    if (status != P_OK) { return status; }
    if (type == 0) {
        if (strcmp(parts[2], "-1") == 0) { second = -1; }
        else { status = planner_number(parts[2], 9999, &second); }
    } else if (type == 1 || type == 3 || type == 5) {
        status = planner_number(parts[2], 9999, &second);
    }
    if (status != P_OK) { return status; }
    if (type == 5) {
        status = planner_number(parts[3], GRAPH_WEIGHT_MAX, &third);
        if (status != P_OK) { return status; }
    }
    if (type == 0) {
        if (p_category(app, first) != P_CLASSES) { return P_DUPLICATE; }
        if (app->category_count == P_CLASSES) { return P_LIMIT; }
        PCategory *row = &app->categories[app->category_count];
        status = label_copy(row->label, parts[3]);
        if (status != P_OK) { return status; }
        row->id = first; row->parent_id = second; app->category_count++;
    } else if (type == 1) {
        if (pending->memberships == RECORD_LIMIT) { return P_LIMIT; }
        size_t i = pending->memberships++;
        pending->record_ids[i] = first; pending->class_ids[i] = second;
    } else if (type == 2) {
        if (p_study(app, first) != GRAPH_MAX) { return P_DUPLICATE; }
        if (app->study_count == GRAPH_MAX) { return P_LIMIT; }
        app->study_ids[app->study_count++] = first;
    } else if (type == 3) {
        if (pending->dependencies == GRAPH_ARCS) { return P_LIMIT; }
        pending->dependency[pending->dependencies++] = (PendingEdge){first,second,1};
    } else if (type == 4) {
        if (p_place(app, first) != GRAPH_MAX) { return P_DUPLICATE; }
        if (app->place_count == GRAPH_MAX) { return P_LIMIT; }
        PPlace *row = &app->places[app->place_count];
        status = label_copy(row->label, parts[2]);
        if (status != P_OK) { return status; }
        row->id = first; app->place_count++;
    } else {
        if (pending->routes == GRAPH_MAX * (GRAPH_MAX + 1) / 2) { return P_LIMIT; }
        pending->route[pending->routes++] = (PendingEdge){first,second,third};
    }
    return P_OK;
}
static PStatus resolve(Planner *app, const Pending *pending)
{
    size_t roots = 0;
    for (size_t i = 0; i < app->category_count; i++) {
        if (app->categories[i].parent_id == -1) { roots++; }
        size_t current = i, steps = 0;
        while (app->categories[current].parent_id != -1) {
            current = p_category(app, app->categories[current].parent_id);
            if (current == P_CLASSES) { return P_MISSING; }
            if (++steps >= app->category_count) { return P_CYCLE; }
        }
    }
    if (app->category_count != 0 && roots != 1) { return P_INVALID; }
    for (size_t i = 0; i < pending->memberships; i++) {
        int position;
        if (open_get(&app->catalog.index, pending->record_ids[i], &position) != H_OK) { return P_MISSING; }
        size_t category = p_category(app, pending->class_ids[i]);
        if (category == P_CLASSES) { return P_MISSING; }
        if (app->membership[position] != P_CLASSES) { return P_DUPLICATE; }
        app->membership[position] = category;
    }
    for (size_t i = 0; i < app->catalog.count; i++) {
        if (app->membership[i] == P_CLASSES) { return P_MISSING; }
    }
    for (size_t i = 0; i < app->study_count; i++) {
        IRecord row;
        if (indexed_find(&app->catalog, app->study_ids[i], &row) != H_OK) { return P_MISSING; }
    }
    (void)graph_init(&app->study, app->study_count, 1);
    (void)graph_init(&app->roads, app->place_count, 0);
    for (size_t i = 0; i < pending->dependencies; i++) {
        PendingEdge edge = pending->dependency[i];
        size_t from = p_study(app, edge.from), to = p_study(app, edge.to);
        if (from == GRAPH_MAX || to == GRAPH_MAX) { return P_MISSING; }
        PStatus status = p_graph_status(graph_add(&app->study, from, to, edge.weight));
        if (status != P_OK) { return status; }
    }
    Topological order;
    PStatus status = p_graph_status(graph_topological(&app->study, &order));
    if (status != P_OK) { return status; }
    for (size_t i = 0; i < pending->routes; i++) {
        PendingEdge edge = pending->route[i];
        size_t from = p_place(app, edge.from), to = p_place(app, edge.to);
        if (from == GRAPH_MAX || to == GRAPH_MAX) { return P_MISSING; }
        status = p_graph_status(graph_add(&app->roads, from, to, edge.weight));
        if (status != P_OK) { return status; }
    }
    return P_OK;
}
PStatus p_relations(FILE *in, Planner *candidate)
{
    Pending pending = {0};
    char line[P_LINE]; int end, phase = -1;
    PStatus status = line_read(in, line, &end);
    if (status != P_OK) { return status; }
    if (end || strcmp(line, "DSA-RELATIONS-1") != 0) { return P_FORMAT; }
    for (;;) {
        status = line_read(in, line, &end);
        if (status != P_OK || end) { break; }
        char *parts[4]; size_t count;
        status = fields(line, parts, &count);
        if (status != P_OK) { return status; }
        if (strlen(parts[0]) != 1) { return P_FORMAT; }
        static const char tags[] = "CMSDPR";
        const char *tag = strchr(tags, parts[0][0]);
        if (tag == NULL) { return P_FORMAT; }
        int type = (int)(tag - tags);
        static const size_t expected[] = {4,3,2,3,3,4};
        if (type < phase || count != expected[type]) { return P_FORMAT; }
        phase = type;
        status = append(candidate, &pending, type, parts);
        if (status != P_OK) { return status; }
    }
    return status == P_OK ? resolve(candidate, &pending) : status;
}
