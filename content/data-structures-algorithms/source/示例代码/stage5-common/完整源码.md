```c
        return 1;

        printf("%s%zu", i == 0 ? "" : ",", pi[i]);
    }
    printf("\nnaive found=%d index=%zu\nkmp found=%d index=%zu\n", a.found, a.index, b.found, b.index);
```

```c
#include "matching.h"

static MStatus validate(const char *text, size_t n, const char *pattern, size_t m, const Match *out)
{
    if (out == NULL || (n != 0 && text == NULL) || (m != 0 && pattern == NULL)) {
        return M_INVALID;
    }
    if (n > MATCH_TEXT_MAX || m > MATCH_PATTERN_MAX) {
        return M_LENGTH;
    }
    return M_OK;
}
MStatus prefix_build(const char *pattern, size_t m, size_t *pi, size_t capacity)
{
    if (m != 0 && (pattern == NULL || pi == NULL)) {
        return M_INVALID;
    }
    if (m > MATCH_PATTERN_MAX || capacity < m) {
        return M_LENGTH;
    }
    if (m == 0) {
        return M_OK;
    }
    pi[0] = 0;
    size_t j = 0;
    for (size_t i = 1; i < m; i++) {
        while (j > 0 && pattern[i] != pattern[j]) {
            j = pi[j - 1];
        }
        if (pattern[i] == pattern[j]) {
            j++;
        }
        pi[i] = j;
    }
    return M_OK;
}
MStatus naive_match(const char *text, size_t n, const char *pattern, size_t m, Match *out)
{
    MStatus status = validate(text, n, pattern, m, out);
    if (status != M_OK) {
        return status;
    }
    Match result = {0, n};
    if (m <= n) {
        for (size_t start = 0; start <= n - m; start++) {
            size_t j = 0;
            while (j < m && text[start + j] == pattern[j]) {
                j++;
            }
            if (j == m) {
                result.found = 1;
                result.index = start;
                break;
            }
        }
    }
    *out = result;
    return M_OK;
}
MStatus kmp_match(const char *text, size_t n, const char *pattern, size_t m, Match *out)
{
    MStatus status = validate(text, n, pattern, m, out);
    if (status != M_OK) {
        return status;
    }
    Match result = {0, n};
    if (m == 0) {
        result.found = 1;
        result.index = 0;
    }
    else if (m <= n) {
        size_t pi[MATCH_PATTERN_MAX];
        (void)prefix_build(pattern, m, pi, MATCH_PATTERN_MAX);
        size_t j = 0;
        for (size_t i = 0; i < n; i++) {
            while (j > 0 && text[i] != pattern[j]) {
                j = pi[j - 1];
            }
            if (text[i] == pattern[j]) {
                j++;
            }
            if (j == m) {
                result.found = 1;
                result.index = i + 1 - m;
                break;
            }
        }
    }
    *out = result;
    return M_OK;
}
```

```c
#include "chaining.h"
#include <stdlib.h>

static HNode *find_node(const ChainMap *map, int key)
{
    HNode *node = map->buckets[(size_t)key % CHAIN_BUCKETS];
    while (node != NULL && node->key != key) {
        node = node->next;
    }
    return node;
}
void chain_init(ChainMap *map)
{
    for (size_t i = 0; i < CHAIN_BUCKETS; i++) {
        map->buckets[i] = NULL;
    }
    map->count = 0;
}
void chain_clear(ChainMap *map)
{
    for (size_t i = 0; i < CHAIN_BUCKETS; i++) {
        HNode *node = map->buckets[i];
        while (node != NULL) {
            HNode *next = node->next;
            free(node);
            node = next;
        }
    }
    chain_init(map);
}
HStatus chain_put(ChainMap *map, int key, int value)
{
    if (key < 0 || key > H_KEY_MAX) {
        return H_INVALID;
    }
    if (find_node(map, key) != NULL) {
        return H_DUPLICATE;
    }
    if (map->count == H_MAX_ITEMS) {
        return H_FULL;
    }
    HNode *node = malloc(sizeof *node);
    if (node == NULL) {
        return H_MEMORY;
    }
    size_t bucket = (size_t)key % CHAIN_BUCKETS;
    node->key = key;
    node->value = value;
    node->next = map->buckets[bucket];
    map->buckets[bucket] = node;
    map->count++;
    return H_OK;
}
HStatus chain_get(const ChainMap *map, int key, int *out)
{
    if (key < 0 || key > H_KEY_MAX || out == NULL) {
        return H_INVALID;
    }
    const HNode *node = find_node(map, key);
    if (node == NULL) {
        return H_MISSING;
    }
    *out = node->value;
    return H_OK;
}
HStatus chain_set(ChainMap *map, int key, int value)
{
    if (key < 0 || key > H_KEY_MAX) {
        return H_INVALID;
    }
    HNode *node = find_node(map, key);
    if (node == NULL) {
        return H_MISSING;
    }
    node->value = value;
    return H_OK;
}
HStatus chain_erase(ChainMap *map, int key)
{
    if (key < 0 || key > H_KEY_MAX) {
        return H_INVALID;
    }
    size_t bucket = (size_t)key % CHAIN_BUCKETS;
    HNode *previous = NULL, *node = map->buckets[bucket];
    while (node != NULL && node->key != key) {
        previous = node;
        node = node->next;
    }
    if (node == NULL) {
        return H_MISSING;
    }
    if (previous == NULL) {
        map->buckets[bucket] = node->next;
    }
    else {
        previous->next = node->next;
    }
    free(node);
    map->count--;
    return H_OK;
}
```

## 29-open-addressing/open_map.c

完整文件：[open_map.c](../29-open-addressing/open_map.c)。

```c
#include "open_map.h"
#include <stdlib.h>

static size_t locate(const OpenMap *map, int key)
{
    if (map->capacity == 0) {
        return 0;
    }
    size_t start = (size_t)key % map->capacity;
    for (size_t step = 0; step < map->capacity; step++) {
        size_t i = (start + step) % map->capacity;
        if (map->slots[i].state == SLOT_EMPTY) {
            break;
        }
        if (map->slots[i].state == SLOT_LIVE && map->slots[i].key == key) {
            return i;
        }
    }
    return map->capacity;
}
static void place(OpenMap *map, int key, int value)
{
    size_t start = (size_t)key % map->capacity;
    for (size_t step = 0; step < map->capacity; step++) {
        size_t i = (start + step) % map->capacity;
        if (map->slots[i].state != SLOT_LIVE) {
            if (map->slots[i].state == SLOT_EMPTY) {
                map->used++;
            }
            map->slots[i].state = SLOT_LIVE;
            map->slots[i].key = key;
            map->slots[i].value = value;
            map->count++;
            return;
        }
    }
}
void open_init(OpenMap *map)
{
    map->slots = NULL;
    map->capacity = map->count = map->used = 0;
}
void open_clear(OpenMap *map)
{
    free(map->slots);
    open_init(map);
}
HStatus open_rehash(OpenMap *map, size_t capacity)
{
    if ((capacity != 8 && capacity != 16 && capacity != 32 && capacity != 64) ||
        map->count > capacity * 3 / 4) {
        return H_INVALID;
    }
    HSlot *slots = calloc(capacity, sizeof *slots);
    if (slots == NULL) {
        return H_MEMORY;
    }
    OpenMap next = {slots, capacity, 0, 0};
    for (size_t i = 0; i < map->capacity; i++) {
        if (map->slots[i].state == SLOT_LIVE) {
            place(&next, map->slots[i].key, map->slots[i].value);
        }
    }
    free(map->slots);
    *map = next;
    return H_OK;
}
HStatus open_put(OpenMap *map, int key, int value)
{
    if (key < 0 || key > H_KEY_MAX) {
        return H_INVALID;
    }
    if (locate(map, key) != map->capacity) {
        return H_DUPLICATE;
    }
    if (map->count == H_MAX_ITEMS) {
        return H_FULL;
    }
    size_t target = map->capacity;
    if (target == 0) {
        target = 8;
    }
    else if (map->count + 1 > target * 3 / 4) {
        target *= 2;
    }
    if (target != map->capacity || map->used + 1 > target * 3 / 4) {
        HStatus status = open_rehash(map, target);
        if (status != H_OK) {
            return status;
        }
    }
    place(map, key, value);
    return H_OK;
}
HStatus open_get(const OpenMap *map, int key, int *out)
{
    if (key < 0 || key > H_KEY_MAX || out == NULL) {
        return H_INVALID;
    }
    size_t i = locate(map, key);
    if (i == map->capacity) {
        return H_MISSING;
    }
    *out = map->slots[i].value;
    return H_OK;
}
HStatus open_set(OpenMap *map, int key, int value)
{
    if (key < 0 || key > H_KEY_MAX) {
        return H_INVALID;
    }
    size_t i = locate(map, key);
    if (i == map->capacity) {
        return H_MISSING;
    }
    map->slots[i].value = value;
    return H_OK;
}
HStatus open_erase(OpenMap *map, int key)
{
    if (key < 0 || key > H_KEY_MAX) {
        return H_INVALID;
    }
    size_t i = locate(map, key);
    if (i == map->capacity) {
        return H_MISSING;
    }
    map->slots[i].state = SLOT_DELETED;
    map->count--;
    return H_OK;
}
```

```c
#include "catalog.h"
#include <string.h>

static int valid_record(const IRecord *record)
{
    if (record->id < 0 || record->id > H_KEY_MAX || record->minutes < 0 || record->minutes > 1440) {
        return 0;
    }
    for (size_t i = 0; i < TITLE_BYTES; i++) {
        unsigned char byte = (unsigned char)record->title[i];
        if (byte == 0) {
            return i != 0;
        }
        if (byte < 32 || byte == 127) {
            return 0;
        }
    }
    return 0;
}
void indexed_init(IndexedCatalog *catalog)
{
    catalog->count = 0;
    open_init(&catalog->index);
}
void indexed_clear(IndexedCatalog *catalog)
{
    open_clear(&catalog->index);
    catalog->count = 0;
}
HStatus indexed_replace(IndexedCatalog *catalog, const IRecord *records, size_t count)
{
    if ((count != 0 && records == NULL) || count > H_MAX_ITEMS) {
        return H_INVALID;
    }
    IRecord candidate[H_MAX_ITEMS];
    OpenMap next;
    open_init(&next);
    for (size_t i = 0; i < count; i++) {
        if (!valid_record(&records[i])) {
            open_clear(&next);
            return H_INVALID;
        }
        candidate[i] = records[i];
        HStatus status = open_put(&next, candidate[i].id, (int)i);
        if (status != H_OK) {
            open_clear(&next);
            return status;
        }
    }
    for (size_t i = 0; i < count; i++) {
        catalog->records[i] = candidate[i];
    }
    open_clear(&catalog->index);
    catalog->index = next;
    catalog->count = count;
    return H_OK;
}
HStatus indexed_add(IndexedCatalog *catalog, const IRecord *record)
{
    if (record == NULL || !valid_record(record)) {
        return H_INVALID;
    }
    int position;
    if (open_get(&catalog->index, record->id, &position) == H_OK) {
        return H_DUPLICATE;
    }
    if (catalog->count == H_MAX_ITEMS) {
        return H_FULL;
    }
    IRecord candidate[H_MAX_ITEMS];
    for (size_t i = 0; i < catalog->count; i++) {
        candidate[i] = catalog->records[i];
    }
    candidate[catalog->count] = *record;
    return indexed_replace(catalog, candidate, catalog->count + 1);
}
HStatus indexed_remove(IndexedCatalog *catalog, int id)
{
    int position;
    HStatus status = open_get(&catalog->index, id, &position);
    if (status != H_OK) {
        return status;
    }
    IRecord candidate[H_MAX_ITEMS];
    size_t n = 0;
    for (size_t i = 0; i < catalog->count; i++) {
        if (i != (size_t)position) {
            candidate[n++] = catalog->records[i];
        }
    }
    return indexed_replace(catalog, candidate, n);
}
HStatus indexed_find(const IndexedCatalog *catalog, int id, IRecord *out)
{
    if (out == NULL) {
        return H_INVALID;
    }
    int position;
    HStatus status = open_get(&catalog->index, id, &position);
    if (status != H_OK) {
        return status;
    }
    *out = catalog->records[position];
    return H_OK;
}
HStatus indexed_set_minutes(IndexedCatalog *catalog, int id, int minutes)
{
    if (minutes < 0 || minutes > 1440) {
        return H_INVALID;
    }
    int position;
    HStatus status = open_get(&catalog->index, id, &position);
    if (status != H_OK) {
        return status;
    }
    catalog->records[position].minutes = minutes;
    return H_OK;
}
```

```c
int main(int argc, char **argv)

    const char *pattern = argc == 2 ? argv[1] : "C";
```
