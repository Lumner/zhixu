```c
#include "interval.h"
static int earlier(Activity a, Activity b)
{
    if (a.end != b.end) { return a.end < b.end; }
    if (a.start != b.start) { return a.start < b.start; }
    return a.id < b.id;
}
DStatus activity_select(const Activity *items, size_t n, Schedule *out)
{
    if (out == NULL || (n != 0 && items == NULL)) { return D_INVALID; }
    if (n > D_ITEMS) { return D_RANGE; }
    Activity sorted[D_ITEMS];
    for (size_t i = 0; i < n; i++) {
        if (items[i].start > D_LIMIT || items[i].end > D_LIMIT) { return D_RANGE; }
        if (items[i].start >= items[i].end) { return D_INVALID; }
        for (size_t j = 0; j < i; j++) {
            if (items[j].id == items[i].id) { return D_INVALID; }
        }
        sorted[i] = items[i];
    }
    for (size_t i = 1; i < n; i++) {
        Activity saved = sorted[i];
        size_t j = i;
        while (j > 0 && earlier(saved, sorted[j - 1])) {
            sorted[j] = sorted[j - 1]; j--;
        }
        sorted[j] = saved;
    }
    Schedule next = {0};
    unsigned last_end = 0;
    for (size_t i = 0; i < n; i++) {
        if (next.count == 0 || sorted[i].start >= last_end) {
            next.ids[next.count++] = sorted[i].id;
            last_end = sorted[i].end;
        }
    }
    *out = next;
    return D_OK;
}
```

## 49-change/greedy.c

[下载文件](../49-change/greedy.c)

```c
#include "coins.h"
DStatus greedy_change(const unsigned *coins, size_t n, unsigned amount, Change *out)
{
    if (out == NULL || (n != 0 && coins == NULL)) { return D_INVALID; }
    if (n > D_COINS || amount > D_LIMIT) { return D_RANGE; }
    for (size_t i = 0; i < n; i++) {
        if (coins[i] == 0 || coins[i] > D_DENOM) { return D_RANGE; }
        if (i > 0 && coins[i - 1] <= coins[i]) { return D_INVALID; }
    }
    Change next = {0};
    unsigned remaining = amount;
    for (size_t i = 0; i < n; i++) {
        next.evaluations++;
        next.counts[i] = remaining / coins[i];
        remaining %= coins[i];
        next.total += next.counts[i];
    }
    if (remaining != 0) { return D_STUCK; }
    next.possible = 1;
    *out = next;
    return D_OK;
}
```

## 50-subset/subset.c

[下载文件](../50-subset/subset.c)

```c
#include "subset.h"
typedef struct {
    const unsigned *values;
    size_t n;
    unsigned target, mask, suffix[D_ITEMS + 1];
    int pruning;
    SubsetResult *result;
} Search;
static void visit(Search *search, size_t i, unsigned sum)
{
    search->result->nodes++;
    if (search->pruning &&
        (sum > search->target || search->target - sum > search->suffix[i])) {
        search->result->pruned++;
        return;
    }
    if (i == search->n) {
        if (sum == search->target) {
            search->result->masks[search->result->count++] = search->mask;
        }
        return;
    }
    unsigned saved = search->mask;
    search->mask |= 1u << i;
    visit(search, i + 1, sum + search->values[i]);
    search->mask = saved;
    visit(search, i + 1, sum);
}
DStatus subset_sums(const unsigned *values, size_t n, unsigned target,
                    int pruning, SubsetResult *out)
{
    if (out == NULL || (n != 0 && values == NULL) || (pruning != 0 && pruning != 1)) {
        return D_INVALID;
    }
    if (n > D_ITEMS || target > D_SUB_TARGET) { return D_RANGE; }
    for (size_t i = 0; i < n; i++) {
        if (values[i] > D_SUB_VALUE) { return D_RANGE; }
    }
    SubsetResult next = {0};
    Search search = {0};
    search.values = values; search.n = n; search.target = target;
    search.pruning = pruning; search.result = &next;
    for (size_t i = n; i > 0; i--) {
        search.suffix[i - 1] = search.suffix[i] + values[i - 1];
    }
    visit(&search, 0, 0);
    *out = next;
    return D_OK;
}
```

## 50-sudoku/sudoku.c

[下载文件](../50-sudoku/sudoku.c)

```c
#include "sudoku.h"
typedef struct {
    MiniGrid board;
    unsigned rows[MINI_SIDE], cols[MINI_SIDE], boxes[MINI_SIDE];
    SudokuResult *result;
} SudokuSearch;
static void fill(SudokuSearch *search, size_t position)
{
    search->result->nodes++;
    while (position < 16 && search->board.cells[position / 4][position % 4] != 0) {
        position++;
    }
    if (position == 16) {
        if (search->result->solutions == 0) { search->result->first = search->board; }
        search->result->solutions++;
        return;
    }
    size_t row = position / 4, col = position % 4, box = (row / 2) * 2 + col / 2;
    for (unsigned digit = 1; digit <= 4; digit++) {
        unsigned bit = 1u << (digit - 1);
        if ((search->rows[row] | search->cols[col] | search->boxes[box]) & bit) {
            continue;
        }
        search->board.cells[row][col] = (unsigned char)digit;
        search->rows[row] |= bit; search->cols[col] |= bit; search->boxes[box] |= bit;
        fill(search, position + 1);
        search->board.cells[row][col] = 0;
        search->rows[row] &= ~bit; search->cols[col] &= ~bit; search->boxes[box] &= ~bit;
    }
}
DStatus sudoku_solve(const MiniGrid *puzzle, SudokuResult *out)
{
    if (puzzle == NULL || out == NULL) { return D_INVALID; }
    SudokuResult next = {0};
    SudokuSearch search = {0};
    search.board = *puzzle; search.result = &next;
    for (size_t row = 0; row < 4; row++) {
        for (size_t col = 0; col < 4; col++) {
            unsigned digit = puzzle->cells[row][col];
            if (digit > 4) { return D_RANGE; }
            if (digit == 0) { continue; }
            size_t box = (row / 2) * 2 + col / 2;
            unsigned bit = 1u << (digit - 1);
            if ((search.rows[row] | search.cols[col] | search.boxes[box]) & bit) {
                return D_INVALID;
            }
            search.rows[row] |= bit; search.cols[col] |= bit; search.boxes[box] |= bit;
        }
    }
    fill(&search, 0);
    *out = next;
    return D_OK;
}
```

## 51-dp/coins.c

[下载文件](../51-dp/coins.c)

```c
#include "coins.h"
#define NO_CHANGE (D_LIMIT + 1u)
typedef struct {
    const unsigned *coins;
    size_t n, choice[D_LIMIT + 1];
    unsigned best[D_LIMIT + 1];
    unsigned char known[D_LIMIT + 1];
    unsigned long evaluations;
} CoinWork;
static DStatus validate(const unsigned *coins, size_t n, unsigned amount, Change *out)
{
    if (out == NULL || (n != 0 && coins == NULL)) { return D_INVALID; }
    if (n > D_COINS || amount > D_LIMIT) { return D_RANGE; }
    for (size_t i = 0; i < n; i++) {
        if (coins[i] == 0 || coins[i] > D_DENOM) { return D_RANGE; }
        if (i > 0 && coins[i - 1] <= coins[i]) { return D_INVALID; }
    }
    return D_OK;
}
static CoinWork begin(const unsigned *coins, size_t n)
{
    CoinWork work = {0};
    work.coins = coins; work.n = n;
    for (unsigned a = 0; a <= D_LIMIT; a++) {
        work.best[a] = NO_CHANGE; work.choice[a] = D_COINS;
    }
    return work;
}
static unsigned solve(CoinWork *work, unsigned amount)
{
    if (work->known[amount]) { return work->best[amount]; }
    work->known[amount] = 1;
    work->evaluations++;
    if (amount == 0) { work->best[0] = 0; return 0; }
    for (size_t i = 0; i < work->n; i++) {
        if (work->coins[i] > amount) { continue; }
        unsigned child = solve(work, amount - work->coins[i]);
        if (child != NO_CHANGE && child + 1 < work->best[amount]) {
            work->best[amount] = child + 1;
            work->choice[amount] = i;
        }
    }
    return work->best[amount];
}
static Change finish(const CoinWork *work, unsigned amount)
{
    Change next = {0};
    next.evaluations = work->evaluations;
    if (work->best[amount] == NO_CHANGE) { return next; }
    next.possible = 1; next.total = work->best[amount];
    unsigned remaining = amount;
    while (remaining != 0) {
        size_t coin = work->choice[remaining];
        next.counts[coin]++;
        remaining -= work->coins[coin];
    }
    return next;
}
DStatus min_change_memo(const unsigned *coins, size_t n, unsigned amount, Change *out)
{
    DStatus status = validate(coins, n, amount, out);
    if (status != D_OK) { return status; }
    CoinWork work = begin(coins, n);
    (void)solve(&work, amount);
    *out = finish(&work, amount);
    return D_OK;
}
DStatus min_change_table(const unsigned *coins, size_t n, unsigned amount, Change *out)
{
    DStatus status = validate(coins, n, amount, out);
    if (status != D_OK) { return status; }
    CoinWork work = begin(coins, n);
    work.best[0] = 0; work.evaluations = 1;
    for (unsigned a = 1; a <= amount; a++) {
        work.evaluations++;
        for (size_t i = 0; i < n; i++) {
            if (coins[i] > a) { continue; }
            unsigned child = work.best[a - coins[i]];
            if (child != NO_CHANGE && child + 1 < work.best[a]) {
                work.best[a] = child + 1; work.choice[a] = i;
            }
        }
    }
    *out = finish(&work, amount);
    return D_OK;
}
```

```c
#include "knapsack.h"
static DStatus validate(const PackItem *items, size_t n, unsigned budget, const void *out)
{
    if (out == NULL || (n != 0 && items == NULL)) { return D_INVALID; }
    if (n > D_ITEMS || budget > D_LIMIT) { return D_RANGE; }
    for (size_t i = 0; i < n; i++) {
        if (items[i].cost == 0 || items[i].cost > D_LIMIT || items[i].value > D_VALUE) {
            return D_RANGE;
        }
    }
    return D_OK;
}
DStatus pack01_table(const PackItem *items, size_t n, unsigned budget, PackResult *out)
{
    DStatus status = validate(items, n, budget, out);
    if (status != D_OK) { return status; }
    unsigned best[D_ITEMS + 1][D_LIMIT + 1] = {{0}};
    for (size_t i = 1; i <= n; i++) {
        PackItem item = items[i - 1];
        for (unsigned cap = 0; cap <= budget; cap++) {
            best[i][cap] = best[i - 1][cap];
            if (item.cost <= cap) {
                unsigned candidate = best[i - 1][cap - item.cost] + item.value;
                if (candidate > best[i][cap]) { best[i][cap] = candidate; }
            }
        }
    }
    PackResult next = {0};
    next.value = best[n][budget];
    unsigned cap = budget;
    for (size_t i = n; i > 0; i--) {
        if (best[i][cap] != best[i - 1][cap]) {
            next.mask |= 1u << (i - 1);
            next.cost += items[i - 1].cost;
            cap -= items[i - 1].cost;
        }
    }
    *out = next;
    return D_OK;
}
DStatus pack01_rolling(const PackItem *items, size_t n, unsigned budget, unsigned *out)
{
    DStatus status = validate(items, n, budget, out);
    if (status != D_OK) { return status; }
    unsigned best[D_LIMIT + 1] = {0};
    for (size_t i = 0; i < n; i++) {
        for (unsigned cap = budget; cap >= items[i].cost; cap--) {
            unsigned candidate = best[cap - items[i].cost] + items[i].value;
            if (candidate > best[cap]) { best[cap] = candidate; }
        }
    }
    *out = best[budget];
    return D_OK;
}
```
